package com.ztune.app.presentation.settings

import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import com.ztune.app.data.settings.AppSettings
import com.ztune.app.data.settings.ThemeMode
import com.ztune.app.data.transport.DemoScenario
import dagger.hilt.android.lifecycle.HiltViewModel
import javax.inject.Inject
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch

data class SettingsUiState(
    val themeMode: ThemeMode = ThemeMode.SYSTEM,
    val usbBaudRate: Int = 115200,
    val autoReconnect: Boolean = true,
    val reconnectMaxAttempts: Int = 5,
    val reconnectDelayMs: Long = 2_000,
    val demoScenario: DemoScenario = DemoScenario.STABLE
)

@HiltViewModel
class SettingsViewModel @Inject constructor(
    private val settings: AppSettings
) : ViewModel() {

    private val _uiState = MutableStateFlow(SettingsUiState())
    val uiState: StateFlow<SettingsUiState> = _uiState.asStateFlow()

    init {
        viewModelScope.launch {
            settings.themeMode.collect { v -> _uiState.update { it.copy(themeMode = v) } }
        }
        viewModelScope.launch {
            settings.usbBaudRate.collect { v -> _uiState.update { it.copy(usbBaudRate = v) } }
        }
        viewModelScope.launch {
            settings.autoReconnect.collect { v -> _uiState.update { it.copy(autoReconnect = v) } }
        }
        viewModelScope.launch {
            settings.reconnectMaxAttempts.collect { v -> _uiState.update { it.copy(reconnectMaxAttempts = v) } }
        }
        viewModelScope.launch {
            settings.reconnectDelayMs.collect { v -> _uiState.update { it.copy(reconnectDelayMs = v) } }
        }
        viewModelScope.launch {
            settings.demoScenario.collect { v -> _uiState.update { it.copy(demoScenario = v) } }
        }
    }

    fun setThemeMode(mode: ThemeMode) = settings.setThemeMode(mode)
    fun setBaudRate(baud: Int) = settings.setUsbBaudRate(baud)
    fun setAutoReconnect(enabled: Boolean) = settings.setAutoReconnect(enabled)
    fun setMaxAttempts(value: Int) = settings.setReconnectMaxAttempts(value)
    fun setDelayMs(value: Long) = settings.setReconnectDelayMs(value)
    fun setDemoScenario(scenario: DemoScenario) = settings.setDemoScenario(scenario)
}
