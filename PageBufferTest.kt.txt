package com.ztune.app.domain.memory

import com.ztune.app.domain.ini.IniDefinitionParser
import com.ztune.app.domain.ini.model.Endianness
import org.junit.Assert.assertEquals
import org.junit.Assert.assertNull
import org.junit.Assert.assertTrue
import org.junit.Test

class PageBufferTest {

    private val definition = IniDefinitionParser().parse(
        """
        [Constants]
           page = 0
           pageSize = 16
           lim = scalar, U08, 2, "x", 1.0, 0.0, 0.0, 255.0, 0
           bins = array, U08, 4, [4], "y", 1.0, 0.0, 0.0, 255.0, 0
        """.trimIndent()
    )

    @Test
    fun `decode konstanta page dari buffer`() {
        val page = PageBuffer(0, ByteArray(16) { it.toByte() }, definition, Endianness.LITTLE)
        assertEquals(2.0, page.decodeScalar("lim")!!, 1e-9)
        assertEquals(listOf(4.0, 5.0, 6.0, 7.0), page.decodeArray("bins"))
        assertEquals(setOf("lim", "bins"), page.decodeAllScalars().keys)
    }

    @Test
    fun `konstanta yang tidak muat di buffer dikecualikan fail-closed`() {
        // pageSize di INI = 16, tapi buffer nyata hanya 4 byte → bins (@4) tidak layak
        val page = PageBuffer(0, ByteArray(4), definition, Endianness.LITTLE)
        assertTrue(page.pageConstants.none { it.name == "bins" })
        assertNull(page.decodeArray("bins"))
        assertEquals(2.0, page.decodeScalar("lim")!!, 1e-9)
    }

    @Test
    fun `nama yang tidak dikenal mengembalikan null`() {
        val page = PageBuffer(0, ByteArray(16), definition, Endianness.LITTLE)
        assertNull(page.decodeScalar("tidakAda"))
    }
}
