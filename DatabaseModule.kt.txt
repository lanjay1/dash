package com.ztune.app.data.di

import android.content.Context
import androidx.room.Room
import com.ztune.app.data.db.DatalogDao
import com.ztune.app.data.db.ZTuneDatabase
import dagger.Module
import dagger.Provides
import dagger.hilt.InstallIn
import dagger.hilt.android.qualifiers.ApplicationContext
import dagger.hilt.components.SingletonComponent
import javax.inject.Singleton

@Module
@InstallIn(SingletonComponent::class)
object DatabaseModule {

    @Provides
    @Singleton
    fun provideDatabase(@ApplicationContext context: Context): ZTuneDatabase =
        Room.databaseBuilder(context, ZTuneDatabase::class.java, "ztune.db").build()

    @Provides
    fun provideDatalogDao(db: ZTuneDatabase): DatalogDao = db.datalogDao()
}
