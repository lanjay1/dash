package com.ztune.app.domain.protocol

/** Satu frame respons ECU (flag + payload, CRC sudah/telah diverifikasi). */
data class MsResponseFrame(val flag: Int, val payload: ByteArray) {
    override fun equals(other: Any?): Boolean =
        other is MsResponseFrame && other.flag == flag && other.payload.contentEquals(payload)
    override fun hashCode(): Int = flag * 31 + payload.contentHashCode()
    override fun toString(): String = "MsResponseFrame(flag=0x%02X, payload=${payload.size} byte)".format(flag)
}

/**
 * Codec envelope MegaSquirt-style "msEnvelope_1.0" (kompatibel rusEFI):
 *
 *   Request  (app → ECU): [Size: 2 BE] [Payload]             [CRC32: 4 BE]
 *   Response (ECU → app): [Size: 2 BE] [Flag: 1] [Payload]   [CRC32: 4 BE]
 *
 *   Size  = panjang SELURUH paket, termasuk field Size (2) dan CRC (4).
 *   CRC32 = standar reflected 0xEDB88320, dihitung atas Payload (request)
 *           atau Flag+Payload (response) — TIDAK termasuk Size & CRC itu sendiri.
 *
 * Endianness command word DI DALAM payload (page/offset/count) = BIG di wire
 * (kompatibel MS/rusEFI). DATA memori internal ECU = LITTLE (per INI — FASE 3).
 */
object MsEnvelopeCodec {
    const val RESPONSE_HEADER_SIZE = 3   // Size(2) + Flag(1)
    const val REQUEST_HEADER_SIZE = 2    // Size(2)
    const val CRC_SIZE = 4

    const val FLAG_OK = 0x00
    const val FLAG_REALTIME = 0x01

    fun encodeRequest(payload: ByteArray): ByteArray {
        val size = REQUEST_HEADER_SIZE + payload.size + CRC_SIZE
        require(size <= 0xFFFF) { "Paket request terlalu besar: $size byte" }
        val out = ByteArray(size)
        out[0] = (size ushr 8).toByte()
        out[1] = size.toByte()
        payload.copyInto(out, REQUEST_HEADER_SIZE)
        writeCrc(out, REQUEST_HEADER_SIZE + payload.size, Crc32.compute(payload))
        return out
    }

    fun encodeResponse(flag: Int, payload: ByteArray): ByteArray {
        val size = RESPONSE_HEADER_SIZE + payload.size + CRC_SIZE
        require(size <= 0xFFFF) { "Paket response terlalu besar: $size byte" }
        val out = ByteArray(size)
        out[0] = (size ushr 8).toByte()
        out[1] = size.toByte()
        out[2] = flag.toByte()
        payload.copyInto(out, RESPONSE_HEADER_SIZE)
        val crcInput = ByteArray(1 + payload.size)
        crcInput[0] = flag.toByte()
        payload.copyInto(crcInput, 1)
        writeCrc(out, RESPONSE_HEADER_SIZE + payload.size, Crc32.compute(crcInput))
        return out
    }

    fun writeCrc(dest: ByteArray, offset: Int, crc: Int) {
        dest[offset] = (crc ushr 24).toByte()
        dest[offset + 1] = (crc ushr 16).toByte()
        dest[offset + 2] = (crc ushr 8).toByte()
        dest[offset + 3] = crc.toByte()
    }

    fun readCrc(src: ByteArray, offset: Int): Long =
        ((src[offset].toLong() and 0xFF) shl 24) or
            ((src[offset + 1].toLong() and 0xFF) shl 16) or
            ((src[offset + 2].toLong() and 0xFF) shl 8) or
            (src[offset + 3].toLong() and 0xFF)

    fun readU16BE(src: ByteArray, offset: Int): Int =
        ((src[offset].toInt() and 0xFF) shl 8) or (src[offset + 1].toInt() and 0xFF)

    fun writeU16BE(dest: ByteArray, offset: Int, value: Int) {
        dest[offset] = ((value ushr 8) and 0xFF).toByte()
        dest[offset + 1] = (value and 0xFF).toByte()
    }
}
