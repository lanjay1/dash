@Composable
fun TuneEditorScreen(
    onNavigateToConnection: () -> Unit,
    modifier: Modifier = Modifier,
    viewModel: TuneEditorViewModel = hiltViewModel()
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    TuneEditorContent(
        uiState = uiState,
        onSelectTable = viewModel::selectTable,
        onLoadPage = viewModel::loadPageIfNeeded,
        onZoomIn = viewModel::zoomIn,
        onZoomOut = viewModel::zoomOut,
        onToggleSelectMode = viewModel::toggleSelectMode,
        onClearSelection = viewModel::clearSelection,
        onToggleCell = viewModel::toggleCell,
        onEditCell = viewModel::editCell,
        onInterpRows = viewModel::interpolateRowWise,
        onInterpCols = viewModel::interpolateColumnWise,
        onUndo = viewModel::undo,
        onRedo = viewModel::redo,
        onRequestWrite = viewModel::requestWrite,
        onConfirmWrite = viewModel::confirmWrite,
        onDismissWrite = viewModel::dismissWrite,
        onClearMessage = viewModel::clearMessage,
        onOpenSessionPicker = viewModel::openSessionPicker,
        onDismissSessionPicker = viewModel::dismissSessionPicker,
        onAnalyzeSession = viewModel::analyzeSession,
        onApplyAutoTune = viewModel::applyAutoTune,
        onDismissAutoTune = viewModel::dismissAutoTune,
        onNavigateToConnection = onNavigateToConnection,
        modifier = modifier
    )
}

@Composable
private fun TuneEditorContent(
    uiState: TuneEditorUiState,
    onSelectTable: (Int) -> Unit,
    onLoadPage: () -> Unit,
    onZoomIn: () -> Unit,
    onZoomOut: () -> Unit,
    onToggleSelectMode: () -> Unit,
    onClearSelection: () -> Unit,
    onToggleCell: (Int, Int) -> Unit,
    onEditCell: (Int, Int, Double) -> Unit,
    onInterpRows: () -> Unit,
    onInterpCols: () -> Unit,
    onUndo: () -> Unit,
    onRedo: () -> Unit,
    onRequestWrite: () -> Unit,
    onConfirmWrite: () -> Unit,
    onDismissWrite: () -> Unit,
    onClearMessage: () -> Unit,
    onOpenSessionPicker: () -> Unit,
    onDismissSessionPicker: () -> Unit,
    onAnalyzeSession: (Long) -> Unit,
    onApplyAutoTune: () -> Unit,
    onDismissAutoTune: () -> Unit,
    onNavigateToConnection: () -> Unit,
    modifier: Modifier = Modifier
) {
    Column(
        modifier = modifier
            .fillMaxSize()
            .verticalScroll(rememberScrollState())
            .padding(16.dp),
        verticalArrangement = Arrangement.spacedBy(12.dp)
    ) {
        Text(stringResource(R.string.tune_title), style = MaterialTheme.typography.headlineMedium)

        Row(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
            uiState.tables.forEachIndexed { index, table ->
                FilterChip(
                    selected = index == uiState.selectedIndex,
                    onClick = { onSelectTable(index) },
                    label = { Text(table.title) }
                )
            }
        }

        uiState.message?.let { message ->
            Card(
                modifier = Modifier.fillMaxWidth(),
                colors = CardDefaults.cardColors(
                    containerColor = if (message.startsWith("Tulis") || message.startsWith("Saran")) {
                        MaterialTheme.colorScheme.primaryContainer
                    } else {
                        MaterialTheme.colorScheme.errorContainer
                    }
                )
            ) {
                Row(
                    modifier = Modifier.padding(horizontal = 16.dp, vertical = 8.dp),
                    verticalAlignment = Alignment.CenterVertically
                ) {
                    Text(message, modifier = Modifier.weight(1f), style = MaterialTheme.typography.bodyMedium)
                    TextButton(onClick = onClearMessage) { Text("×") }
                }
            }
        }

        val connected = uiState.connectionStatus == EcuConnectionStatus.CONNECTED
        when {
            !connected -> NotConnectedCard(onNavigateToConnection)
            uiState.isLoading -> Text(stringResource(R.string.tune_loading))
            uiState.grid == null -> Column(verticalArrangement = Arrangement.spacedBy(8.dp)) {
                Text(stringResource(R.string.tune_not_loaded))
                Button(onClick = onLoadPage) { Text(stringResource(R.string.tune_load_page)) }
            }
            else -> EditorSection(
                uiState = uiState,
                onZoomIn = onZoomIn,
                onZoomOut = onZoomOut,
                onToggleSelectMode = onToggleSelectMode,
                onClearSelection = onClearSelection,
                onToggleCell = onToggleCell,
                onEditCell = onEditCell,
                onInterpRows = onInterpRows,
                onInterpCols = onInterpCols,
                onUndo = onUndo,
                onRedo = onRedo,
                onRequestWrite = onRequestWrite
            )
        }

        if (connected && uiState.grid != null) {
            AutoTuneSection(
                suggestion = uiState.autoTune,
                isAnalyzing = uiState.isAnalyzing,
                enabled = !uiState.isWriting,
                onPickSession = onOpenSessionPicker,
                onApply = onApplyAutoTune,
                onDismiss = onDismissAutoTune
            )
        }

        uiState.pendingWrite?.let { pending ->
            AlertDialog(
                onDismissRequest = onDismissWrite,
                title = { Text(stringResource(R.string.tune_write_confirm_title)) },
                text = {
                    Text(
                        stringResource(
                            R.string.tune_write_confirm_body,
                            pending.tableTitle, pending.page, pending.offset, pending.byteCount
                        )
                    )
                },
                confirmButton = {
                    Button(onClick = onConfirmWrite) { Text(stringResource(R.string.tune_write_confirm_ok)) }
                },
                dismissButton = {
                    OutlinedButton(onClick = onDismissWrite) { Text(stringResource(R.string.tune_write_cancel)) }
                }
            )
        }

        if (uiState.isWriting) {
            Text(stringResource(R.string.tune_writing), color = MaterialTheme.colorScheme.primary)
        }
        Spacer(Modifier.height(8.dp))
    }
}

@Composable
private fun NotConnectedCard(onNavigateToConnection: () -> Unit) {
    Card(modifier = Modifier.fillMaxWidth()) {
        Column(
            modifier = Modifier.padding(16.dp),
            verticalArrangement = Arrangement.spacedBy(8.dp)
        ) {
            Text(stringResource(R.string.tune_not_connected), style = MaterialTheme.typography.titleMedium)
            Text(
                stringResource(R.string.tune_connect_hint),
                style = MaterialTheme.typography.bodyMedium,
                color = MaterialTheme.colorScheme.onSurfaceVariant
            )
            Button(onClick = onNavigateToConnection) { Text(stringResource(R.string.dash_button_connect)) }
        }
    }
}

@Composable
private fun EditorSection(
    uiState: TuneEditorUiState,
    onZoomIn: () -> Unit,
    onZoomOut: () -> Unit,
    onToggleSelectMode: () -> Unit,
    onClearSelection: () -> Unit,
    onToggleCell: (Int, Int) -> Unit,
    onEditCell: (Int, Int, Double) -> Unit,
    onInterpRows: () -> Unit,
    onInterpCols: () -> Unit,
    onUndo: () -> Unit,
    onRedo: () -> Unit,
    onRequestWrite: () -> Unit
) {
    val grid = uiState.grid ?: return
    var editingCell by remember { mutableStateOf<Pair<Int, Int>?>(null) }

    Row(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
        OutlinedButton(onClick = onUndo, enabled = uiState.canUndo) { Text("↶") }
        OutlinedButton(onClick = onRedo, enabled = uiState.canRedo) { Text("↷") }
        OutlinedButton(onClick = onZoomOut) { Text("−") }
        OutlinedButton(onClick = onZoomIn) { Text("+") }
        FilterChip(
            selected = uiState.selectMode,
            onClick = onToggleSelectMode,
            label = { Text(stringResource(R.string.tune_select_mode)) }
        )
        if (uiState.selected.isNotEmpty()) {
            OutlinedButton(onClick = onClearSelection) { Text("✕ ${uiState.selected.size}") }
        }
    }
    Row(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
        Button(onClick = onInterpRows, enabled = uiState.selected.size >= 2) {
            Text(stringResource(R.string.tune_interp_rows))
        }
        Button(onClick = onInterpCols, enabled = uiState.selected.size >= 2) {
            Text(stringResource(R.string.tune_interp_cols))
        }
    }
    Button(onClick = onRequestWrite, enabled = !uiState.isWriting, modifier = Modifier.fillMaxWidth()) {
        Text(
            if (uiState.dirty) stringResource(R.string.tune_write_dirty)
            else stringResource(R.string.tune_write_clean)
        )
    }

    val cellSize = (52.dp * uiState.zoom).coerceIn(30.dp, 130.dp)
    val textStyle = MaterialTheme.typography.labelMedium.copy(fontFamily = FontFamily.Monospace)
    val showText = cellSize >= 40.dp

    Column(
        modifier = Modifier
            .fillMaxWidth()
            .horizontalScroll(rememberScrollState())
    ) {
        Row {
            Spacer(Modifier.width(52.dp))
            grid.xBins.forEach { x ->
                Box(Modifier.width(cellSize), contentAlignment = Alignment.Center) {
                    Text(formatAxis(x), style = textStyle, color = MaterialTheme.colorScheme.onSurfaceVariant)
                }
            }
        }
        grid.cells.forEachIndexed { r, rowValues ->
            Row {
                Box(
                    Modifier.width(52.dp).height(cellSize),
                    contentAlignment = Alignment.CenterStart
                ) {
                    Text(
                        formatAxis(grid.yBins.getOrNull(r)),
                        style = textStyle,
                        color = MaterialTheme.colorScheme.onSurfaceVariant
                    )
                }
                rowValues.forEachIndexed { c, value ->
                    val selected = CellPos(r, c) in uiState.selected
                    val bg = heatColor(value, grid.heatMin, grid.heatMax)
                    Box(
                        modifier = Modifier
                            .size(cellSize)
                            .padding(1.dp)
                            .background(bg, RoundedCornerShape(4.dp))
                            .border(
                                width = if (selected) 2.dp else 0.dp,
                                color = if (selected) MaterialTheme.colorScheme.primary else Color.Transparent,
                                shape = RoundedCornerShape(4.dp)
                            )
                            .clickable {
                                if (uiState.selectMode) onToggleCell(r, c) else editingCell = r to c
                            },
                        contentAlignment = Alignment.Center
                    ) {
                        if (showText) {
                            Text(
                                formatValue(value, grid.digits),
                                style = textStyle,
                                color = Color.White
                            )
                        }
                    }
                }
            }
        }
    }

    editingCell?.let { (r, c) ->
        val current = grid.cells.getOrNull(r)?.getOrNull(c)
        var text by remember(r, c) { mutableStateOf(formatValue(current, grid.digits)) }
        AlertDialog(
            onDismissRequest = { editingCell = null },
            title = { Text(stringResource(R.string.tune_edit_cell_title, c, r)) },
            text = {
                OutlinedTextField(
                    value = text,
                    onValueChange = { text = it },
                    singleLine = true,
                    label = { Text("${grid.units} [${grid.heatMin}..${grid.heatMax}]") }
                )
            },
            confirmButton = {
                Button(onClick = {
                    text.toDoubleOrNull()?.let { onEditCell(r, c, it) }
                    editingCell = null
                }) { Text(stringResource(R.string.tune_edit_ok)) }
            },
            dismissButton = {
                TextButton(onClick = { editingCell = null }) { Text(stringResource(R.string.tune_edit_cancel)) }
            }
        )
    }
}

@Composable
private fun AutoTuneSection(
    suggestion: AutoTuneSuggestionUi?,
    isAnalyzing: Boolean,
    enabled: Boolean,
    onPickSession: () -> Unit,
    onApply: () -> Unit,
    onDismiss: () -> Unit
) {
    Column(verticalArrangement = Arrangement.spacedBy(8.dp)) {
        OutlinedButton(onClick = onPickSession, enabled = enabled && !isAnalyzing) {
            Text(stringResource(R.string.tune_autotune_button))
        }
        if (isAnalyzing) {
            Text(stringResource(R.string.tune_autotune_analyzing), style = MaterialTheme.typography.bodyMedium)
        }
        suggestion?.let { s ->
            Card(modifier = Modifier.fillMaxWidth()) {
                Column(
                    modifier = Modifier.padding(16.dp),
                    verticalArrangement = Arrangement.spacedBy(8.dp)
                ) {
                    Text(stringResource(R.string.tune_autotune_title), style = MaterialTheme.typography.titleLarge)
                    Text(
                        stringResource(R.string.tune_autotune_summary_fmt, s.usedSamples, s.skippedSamples, s.cells.size),
                        style = MaterialTheme.typography.bodyMedium,
                        color = MaterialTheme.colorScheme.onSurfaceVariant
                    )
                    Text(
                        stringResource(R.string.tune_autotune_warning),
                        style = MaterialTheme.typography.bodySmall,
                        color = MaterialTheme.colorScheme.onSurfaceVariant
                    )
                    if (s.cells.isEmpty()) {
                        Text(stringResource(R.string.tune_autotune_empty), style = MaterialTheme.typography.bodyMedium)
                    } else {
                        Column(
                            modifier = Modifier
                                .fillMaxWidth()
                                .heightIn(max = 220.dp)
                                .verticalScroll(rememberScrollState())
                        ) {
                            s.cells.forEach { cell ->
                                Text(
                                    stringResource(
                                        R.string.tune_autotune_cell_fmt,
                                        cell.row, cell.col,
                                        cell.xLabel, cell.yLabel,
                                        formatValue(cell.currentValue, 1),
                                        formatValue(cell.suggestedValue, 1),
                                        cell.correctionPct,
                                        cell.sampleCount
                                    ),
                                    style = MaterialTheme.typography.bodySmall.copy(fontFamily = FontFamily.Monospace)
                                )
                            }
                        }
                        Row(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
                            Button(onClick = onApply) { Text(stringResource(R.string.tune_autotune_apply)) }
                            OutlinedButton(onClick = onDismiss) { Text(stringResource(R.string.tune_autotune_dismiss)) }
                        }
                    }
                }
            }
        }
    }
}

private fun heatColor(value: Double?, min: Double, max: Double): Color {
    if (value == null || max <= min) return Color(0xFF546E7A)
    val t = ((value - min) / (max - min)).coerceIn(0.0, 1.0)
    val stops = listOf(
        0.0 to Color(0xFF1565C0),
        0.33 to Color(0xFF2E7D32),
        0.66 to Color(0xFFF9A825),
        1.0 to Color(0xFFC62828)
    )
    val seg = (stops.indexOfFirst { it.first > t }).let { if (it <= 0) 1 else it }
    val (t0, c0) = stops[seg - 1]
    val (t1, c1) = stops[seg]
    val f = ((t - t0) / (t1 - t0)).toFloat()
    return lerpColor(c0, c1, f)
}

private fun lerpColor(a: Color, b: Color, f: Float): Color = Color(
    red = a.red + (b.red - a.red) * f,
    green = a.green + (b.green - a.green) * f,
    blue = a.blue + (b.blue - a.blue) * f,
    alpha = 1f
)

private fun formatAxis(v: Double?): String =
    v?.let { if (abs(it) >= 1000) "${(it / 100).toInt() / 10.0}k" else String.format(Locale.US, "%.0f", it) } ?: "--"

private fun formatValue(value: Double?, digits: Int): String =
    value?.let { String.format(Locale.US, "%.${digits.coerceIn(0, 2)}f", it) } ?: "--"
