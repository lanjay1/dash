package com.ztune.app.domain.memory

import com.ztune.app.domain.ini.model.EcuDefinition
import com.ztune.app.domain.ini.model.Endianness
import com.ztune.app.domain.protocol.EcuCommunicationException
import com.ztune.app.domain.protocol.MsProtocolClient

/**
 * Orkestrasi read page utuh via protokol (chunked per blockingFactor dari INI)
 * mengikuti pageSize dari definisi, lalu bungkus jadi [PageBuffer] siap-decode.
 */
class EcuMemoryReader(
    private val client: MsProtocolClient,
    private val definition: EcuDefinition
) {

    /**
     * Baca seluruh page [pageIndex]. blockingFactor 0 = pakai nilai dari INI
     * (fallback konstanta default protokol).
     */
    suspend fun readPage(pageIndex: Int, blockingFactor: Int = 0): PageBuffer {
        val effectiveBlockingFactor = if (blockingFactor > 0) {
            blockingFactor
        } else {
            definition.commands.blockingFactor ?: MsProtocolClient.DEFAULT_BLOCKING_FACTOR
        }
        val page = definition.pages.firstOrNull { it.index == pageIndex }
            ?: throw EcuCommunicationException.InvalidFrame(
                "Page $pageIndex tidak dideklarasikan di definisi INI"
            )
        val size = page.size
            ?: throw EcuCommunicationException.InvalidFrame(
                "pageSize untuk page $pageIndex tidak ada di INI"
            )
        val data = client.readPage(pageIndex, 0, size, effectiveBlockingFactor)
        return PageBuffer(pageIndex, data, definition, definition.meta.endianness ?: Endianness.LITTLE)
    }
}
