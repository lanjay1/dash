package com.ztune.app.domain.protocol

import org.junit.Assert.assertEquals
import org.junit.Assert.assertTrue
import org.junit.Test

class MsEnvelopeFramerTest {

    private fun responseOf(flag: Int, payload: ByteArray): ByteArray =
        MsEnvelopeCodec.encodeResponse(flag, payload)

    private fun asFrame(event: MsEnvelopeFramer.Event): MsResponseFrame =
        (event as MsEnvelopeFramer.Event.Frame).frame

    @Test
    fun `frame utuh didekode menjadi satu event`() {
        val wire = responseOf(0x00, "rusEFI".toByteArray())
        val events = MsEnvelopeFramer().feed(wire)
        assertEquals(1, events.size)
        val frame = asFrame(events[0])
        assertEquals(0x00, frame.flag)
        assertTrue(frame.payload.contentEquals("rusEFI".toByteArray()))
    }

    @Test
    fun `frame payload kosong valid`() {
        val events = MsEnvelopeFramer().feed(responseOf(0x00, ByteArray(0)))
        assertEquals(1, events.size)
        assertEquals(0, asFrame(events[0]).payload.size)
    }

    @Test
    fun `frame terpotong byte-per-byte tetap terassembly`() {
        val wire = responseOf(0x01, ByteArray(50) { it.toByte() })
        val framer = MsEnvelopeFramer()
        val all = mutableListOf<MsEnvelopeFramer.Event>()
        for (b in wire) all += framer.feed(byteArrayOf(b))
        assertEquals(1, all.size)
        assertEquals(50, asFrame(all[0]).payload.size)
    }

    @Test
    fun `dua frame digabung dalam satu feed menghasilkan dua event`() {
        val wire = responseOf(0x00, byteArrayOf(1)) + responseOf(0x83, byteArrayOf(2, 3))
        val events = MsEnvelopeFramer().feed(wire)
        assertEquals(2, events.size)
        assertEquals(0x00, asFrame(events[0]).flag)
        assertEquals(0x83, asFrame(events[1]).flag)
    }

    @Test
    fun `crc rusak menghasilkan error dan framer pulih untuk frame berikutnya`() {
        val good = responseOf(0x00, byteArrayOf(1, 2, 3))
        val bad = responseOf(0x00, byteArrayOf(9, 9, 9)).also { arr ->
            arr[arr.size - 1] = ((arr[arr.size - 1].toInt() + 1) and 0xFF).toByte()
        }
        val events = MsEnvelopeFramer().feed(bad + good)
        assertEquals(2, events.size)
        assertTrue(events[0] is MsEnvelopeFramer.Event.Error)
        assertTrue(events[1] is MsEnvelopeFramer.Event.Frame)
    }

    @Test
    fun `size tidak valid memicu resync dan frame valid setelahnya tetap terdekode`() {
        val good = responseOf(0x00, "OK".toByteArray())
        val junk = byteArrayOf(0xFF, 0xFF, 0x7F) // 0xFFFF & 0xFF7F & 0x7F00 → semuanya di luar batas
        val events = MsEnvelopeFramer().feed(junk + good)
        assertTrue(events.any { it is MsEnvelopeFramer.Event.Error })
        val frame = asFrame(events.last())
        assertTrue(frame.payload.contentEquals("OK".toByteArray()))
    }

    @Test
    fun `encode request roundtrip sesuai spesifikasi envelope`() {
        val payload = MsCommands.readChunk(page = 0, offset = 16, count = 256)
        val wire = MsEnvelopeCodec.encodeRequest(payload)
        // Size = 2 header + 7 payload + 4 crc = 13
        assertEquals(13, MsEnvelopeCodec.readU16BE(wire, 0))
        assertEquals(wire.size, MsEnvelopeCodec.readU16BE(wire, 0))
        // Payload persis di offset 2
        assertTrue(wire.copyOfRange(2, wire.size - 4).contentEquals(payload))
        // CRC di tail = CRC32 payload, big-endian
        val expectedCrc = Crc32.compute(payload).toLong() and 0xFFFFFFFFL
        assertEquals(expectedCrc, MsEnvelopeCodec.readCrc(wire, wire.size - 4))
    }
}
