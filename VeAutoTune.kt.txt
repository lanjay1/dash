package com.ztune.app.domain.autotune

import com.ztune.app.domain.datalog.DatalogSample
import kotlin.math.abs

/** Parameter AutoTune — default konservatif. */
data class AutoTuneConfig(
    /** Sel dengan sampel kurang dari ini tidak diberi saran. */
    val minSamplesPerCell: Int = 5,
    /** Koreksi maksimum per sel (persen) — pengaman utama. */
    val maxCorrectionPct: Double = 25.0,
    /** Error AFR rata-rata di bawah ini → sel dianggap sudah benar. */
    val deadbandAfr: Double = 0.1,
    /** Rentang AFR yang dianggap valid (di luar ini = sensor/transient rusak). */
    val afrValidMin: Double = 9.0,
    val afrValidMax: Double = 20.0
)

/** Saran koreksi satu sel. */
data class AutoTuneCellResult(
    val row: Int,
    val col: Int,
    val currentValue: Double,
    val suggestedValue: Double,
    val correctionPct: Double,
    val meanAfrError: Double,
    val sampleCount: Int
)

data class AutoTuneResult(
    val cells: List<AutoTuneCellResult>,
    val totalSamples: Int,
    val usedSamples: Int,
    val skippedSamples: Int
) {
    val hasSuggestions: Boolean get() = cells.isNotEmpty()
}

/**
 * AutoTune VE dasar: koreksi = rata-rata (AFR_aktual / AFR_target) per sel.
 *
 * Prinsip: VE menentukan bahan bakar; aktual lebih lean (ratio > 1) → VE naik.
 * HANYA valid untuk tabel bahan bakar (VE) — bukan ignition.
 * Batasan (dokumentasi desain): tanpa flag enrichmen firmware (akselerasi,
 * warm-up), transient bisa mempullu sel — log kondisi stabil memberi hasil
 * terbaik. Firmware custom dapat menyediakan channel flag untuk filter nanti.
 */
object VeAutoTune {

    fun analyze(
        samples: List<DatalogSample>,
        afrChannel: String,
        afrTargetChannel: String,
        xAxisChannel: String,
        yAxisChannel: String,
        xBins: List<Double>,
        yBins: List<Double>,
        gridValues: List<List<Double?>>,
        config: AutoTuneConfig = AutoTuneConfig()
    ): AutoTuneResult {
        require(config.minSamplesPerCell >= 1) { "minSamplesPerCell minimal 1" }
        require(config.afrValidMin < config.afrValidMax) { "rentang AFR tidak valid" }
        val cols = xBins.size
        val rows = yBins.size

        class CellAcc {
            var sumRatio = 0.0
            var sumError = 0.0
            var count = 0
        }

        val cells = HashMap<Long, CellAcc>()
        var used = 0
        var skipped = 0

        for (sample in samples) {
            val afr = sample.values[afrChannel]
            val target = sample.values[afrTargetChannel]
            val x = sample.values[xAxisChannel]
            val y = sample.values[yAxisChannel]
            if (afr == null || target == null || x == null || y == null ||
                afr.isNaN() || target.isNaN() || x.isNaN() || y.isNaN()
            ) {
                skipped++
                continue
            }
            if (afr < config.afrValidMin || afr > config.afrValidMax ||
                target < config.afrValidMin || target > config.afrValidMax ||
                target <= 0.0
            ) {
                skipped++
                continue
            }
            val col = nearestBin(xBins, x)
            val row = nearestBin(yBins, y)
            if (col < 0 || row < 0 || col >= cols || row >= rows) {
                skipped++
                continue
            }
            val acc = cells.getOrPut(row.toLong() * cols + col) { CellAcc() }
            acc.sumRatio += afr / target
            acc.sumError += afr - target
            acc.count++
            used++
        }

        val maxFactor = config.maxCorrectionPct / 100.0
        val results = cells.mapNotNull { (key, acc) ->
            if (acc.count < config.minSamplesPerCell) return@mapNotNull null
            val row = (key / cols).toInt()
            val col = (key % cols).toInt()
            val current = gridValues.getOrNull(row)?.getOrNull(col)
            if (current == null || current.isNaN()) return@mapNotNull null
            val meanError = acc.sumError / acc.count
            if (abs(meanError) < config.deadbandAfr) return@mapNotNull null
            val meanRatio = acc.sumRatio / acc.count
            val clamped = meanRatio.coerceIn(1.0 - maxFactor, 1.0 + maxFactor)
            AutoTuneCellResult(
                row = row,
                col = col,
                currentValue = current,
                suggestedValue = current * clamped,
                correctionPct = (clamped - 1.0) * 100.0,
                meanAfrError = meanError,
                sampleCount = acc.count
            )
        }.sortedWith(compareBy({ it.row }, { it.col }))

        return AutoTuneResult(
            cells = results,
            totalSamples = samples.size,
            usedSamples = used,
            skippedSamples = skipped
        )
    }

    /** Indeks bin dengan nilai terdekat; seri → indeks lebih kecil. -1 jika kosong. */
    fun nearestBin(bins: List<Double>, value: Double): Int {
        var best = -1
        var bestDist = Double.MAX_VALUE
        for (i in bins.indices) {
            val d = abs(bins[i] - value)
            if (d < bestDist) {
                bestDist = d
                best = i
            }
        }
        return best
    }
}
