package com.ztune.app.domain.protocol

import kotlinx.coroutines.flow.Flow

/**
 * Abstraksi transport byte mentah ke/dari ECU.
 * Implementasi: USB-serial (UsbSerialTransport), demo in-process (FakeEcuTransport).
 * Bluetooth SPP nanti tinggal mengimplementasikan interface ini —
 * framer & protocol client di atasnya tidak berubah sama sekali.
 */
interface EcuTransport {
    /** Nama manusiawi untuk UI/log (mis. "USB: CP2102", "Demo ECU"). */
    val name: String

    /** Buka koneksi fisik. Throws jika gagal (izin ditolak, device hilang, dsb.). */
    suspend fun connect()

    /** Tutup koneksi & bebaskan resource. Idempotent. Transport single-use. */
    suspend fun disconnect()

    /** Kirim byte mentah; suspend sampai antre/di-flush. */
    suspend fun write(bytes: ByteArray)

    /**
     * Stream byte masuk (chunked, ukuran chunk tidak dijamin).
     * Hanya boleh ada SATU kolektor aktif (MsProtocolClient memaksa lock-step).
     * Flow selesai (complete) saat koneksi ditutup.
     */
    val incoming: Flow<ByteArray>
}
