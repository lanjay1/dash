package com.ztune.app.data.writer

import com.ztune.app.domain.ini.model.EcuDefinition
import com.ztune.app.domain.ini.model.Endianness
import com.ztune.app.domain.memory.ByteOrderReader
import com.ztune.app.domain.protocol.Crc32
import com.ztune.app.domain.protocol.EcuCommunicationException
import com.ztune.app.domain.protocol.MsCommands
import com.ztune.app.domain.protocol.MsProtocolClient

/**
 * Tulis region memori ke ECU dengan rantai verifikasi:
 * 1) 'C' chunked per blockingFactor; 2) verifikasi CRC32 region via 'k'
 * (payload 4 byte: LE dulu, fallback BE — ⚠️ VERIFIKASI-HARDWARE);
 * 3) burn 'B' HANYA jika CRC cocok; 4) readback region dan bandingkan.
 */
class EcuWriter(
    private val client: MsProtocolClient,
    private val definition: EcuDefinition
) {

    data class WriteResult(
        val bytesWritten: Int,
        val crcVerified: Boolean,
        val burned: Boolean,
        val readbackVerified: Boolean
    )

    suspend fun writeRegion(
        page: Int,
        offset: Int,
        data: ByteArray,
        verify: Boolean = true
    ): WriteResult {
        require(data.isNotEmpty()) { "Tidak ada data untuk ditulis" }
        require(offset in 0..0xFFFF) { "offset di luar range: $offset" }
        val blockingFactor = definition.commands.blockingFactor
            ?: MsProtocolClient.DEFAULT_BLOCKING_FACTOR

        // 1) Tulis chunked
        var pos = 0
        while (pos < data.size) {
            val chunk = minOf(blockingFactor, data.size - pos)
            client.request(MsCommands.writeChunk(page, offset + pos, data.copyOfRange(pos, pos + chunk)))
            pos += chunk
        }

        var crcVerified = !verify
        if (verify) {
            // 2) Verifikasi CRC32 region via ECU
            val response = client.request(MsCommands.crc32Region(page, offset, data.size))
            if (response.payload.size < 4) {
                throw EcuCommunicationException.InvalidFrame(
                    "Respons CRC ('k') terlalu pendek: ${response.payload.size} byte"
                )
            }
            val local = Crc32.compute(data).toLong() and 0xFFFFFFFFL
            val asLE = ByteOrderReader.u32(response.payload, 0, Endianness.LITTLE) and 0xFFFFFFFFL
            val asBE = ByteOrderReader.u32(response.payload, 0, Endianness.BIG) and 0xFFFFFFFFL
            crcVerified = local == asLE || local == asBE
            if (!crcVerified) {
                throw EcuCommunicationException.InvalidFrame(
                    "Verifikasi CRC region gagal (lokal 0x%08X, ECU LE 0x%08X / BE 0x%08X)"
                        .format(local, asLE, asBE)
                )
            }
        }

        // 3) Burn
        client.request(MsCommands.burn(page))

        // 4) Readback
        val readback = client.readPage(page, offset, data.size, blockingFactor)
        val readbackVerified = readback.contentEquals(data)

        return WriteResult(
            bytesWritten = data.size,
            crcVerified = crcVerified,
            burned = true,
            readbackVerified = readbackVerified
        )
    }
}
