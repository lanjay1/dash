package com.ztune.app.domain.autotune

import com.ztune.app.domain.datalog.DatalogSample
import org.junit.Assert.assertEquals
import org.junit.Assert.assertFalse
import org.junit.Assert.assertTrue
import org.junit.Test

class VeAutoTuneTest {

    private val xBins = listOf(1000.0, 2000.0, 3000.0, 4000.0)
    private val yBins = listOf(20.0, 40.0, 60.0, 80.0)
    private val grid = List(4) { List(4) { 50.0 } }
    private val cfg = AutoTuneConfig(
        minSamplesPerCell = 5, maxCorrectionPct = 25.0, deadbandAfr = 0.1
    )

    private fun sample(t: Long, rpm: Double, clt: Double, afr: Double, target: Double) =
        DatalogSample(t, mapOf("RPM" to rpm, "CLT" to clt, "AFR" to afr, "AFRT" to target))

    private fun analyze(samples: List<DatalogSample>) = VeAutoTune.analyze(
        samples, "AFR", "AFRT", "RPM", "CLT", xBins, yBins, grid, cfg
    )

    @Test
    fun `tune sempurna tidak menghasilkan saran`() {
        val samples = (0 until 40).map {
            sample(it.toLong(), 1500.0 + it * 100 % 3000, 30.0 + it % 50, 14.7, 14.7)
        }
        val result = analyze(samples)
        assertTrue(result.usedSamples > 0)
        assertFalse(result.hasSuggestions)
    }

    @Test
    fun `sel lean 8_8 persen mendapat saran naik sesuai formula`() {
        val samples = (0 until 6).map { sample(it.toLong(), 1900.0, 55.0, 16.0, 14.7) } +
            (0 until 6).map { sample(100 + it.toLong(), 2900.0, 35.0, 14.7, 14.7) }
        val result = analyze(samples)
        assertEquals(12, result.usedSamples)
        assertEquals(1, result.cells.size)
        val cell = result.cells[0]
        assertEquals(2, cell.row) // CLT 55 → bin 60 (row 2)
        assertEquals(1, cell.col) // RPM 1900 → bin 2000 (col 1)
        assertEquals(6, cell.sampleCount)
        assertEquals(50.0 * 16.0 / 14.7, cell.suggestedValue, 1e-9)
        assertEquals((16.0 / 14.7 - 1.0) * 100.0, cell.correctionPct, 1e-9)
        assertEquals(1.3, cell.meanAfrError, 1e-9)
    }

    @Test
    fun `koreksi di-clamp ke batas maksimum`() {
        val samples = (0 until 6).map { sample(it.toLong(), 1900.0, 55.0, 20.0, 14.7) } // ratio 1.36
        val result = analyze(samples)
        assertEquals(1, result.cells.size)
        assertEquals(25.0, result.cells[0].correctionPct, 1e-9)
        assertEquals(62.5, result.cells[0].suggestedValue, 1e-9)
    }

    @Test
    fun `sel dengan sampel kurang dari minimum diabaikan`() {
        val samples = (0 until 4).map { sample(it.toLong(), 1900.0, 55.0, 16.0, 14.7) }
        val result = analyze(samples)
        assertEquals(4, result.usedSamples)
        assertFalse(result.hasSuggestions)
    }

    @Test
    fun `sampel tidak valid dilewati dan terhitung`() {
        val samples = listOf(
            sample(0, 1900.0, 55.0, Double.NaN, 14.7),
            sample(1, 1900.0, 55.0, 8.0, 14.7),      // di bawah afrValidMin
            sample(2, 1900.0, 55.0, 16.0, 0.0),      // target ≤ 0
            DatalogSample(3, mapOf("RPM" to 1900.0, "CLT" to 55.0, "AFR" to 16.0)) // target hilang
        )
        val result = analyze(samples)
        assertEquals(4, result.skippedSamples)
        assertEquals(0, result.usedSamples)
        assertEquals(4, result.totalSamples)
        assertFalse(result.hasSuggestions)
    }

    @Test
    fun `pemetaan bin terdekat dan seri jatuh ke indeks kecil`() {
        val samples = (0 until 6).map { sample(it.toLong(), 1500.0, 55.0, 16.0, 14.7) } // 1500 seri
        val result = analyze(samples)
        assertEquals(0, result.cells[0].col)
    }

    @Test
    fun `error di bawah deadband tanpa saran`() {
        val samples = (0 until 6).map { sample(it.toLong(), 1900.0, 55.0, 14.75, 14.7) }
        val result = analyze(samples)
        assertEquals(6, result.usedSamples)
        assertFalse(result.hasSuggestions)
    }

    @Test
    fun `sel dengan nilai grid NaN tidak diberi saran`() {
        val gridWithNaN = grid.mapIndexed { r, row ->
            row.mapIndexed { c, v -> if (r == 2 && c == 1) Double.NaN else v }
        }
        val samples = (0 until 6).map { sample(it.toLong(), 1900.0, 55.0, 16.0, 14.7) }
        val result = VeAutoTune.analyze(
            samples, "AFR", "AFRT", "RPM", "CLT", xBins, yBins, gridWithNaN, cfg
        )
        assertFalse(result.hasSuggestions)
    }
}
