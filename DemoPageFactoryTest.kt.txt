package com.ztune.app.domain.protocol

import com.ztune.app.data.transport.FakeEcuTransport
import com.ztune.app.domain.ini.BundledEcuDefinition
import com.ztune.app.domain.ini.IniDefinitionParser
import com.ztune.app.domain.memory.EcuMemoryReader
import kotlinx.coroutines.runBlocking
import org.junit.Assert.assertEquals
import org.junit.Test

class DemoPageFactoryTest {

    private val definition = IniDefinitionParser().parse(BundledEcuDefinition.INI)

    @Test
    fun `page0 terdecode sesuai formula factory dan layout ini`() = runBlocking {
        val ecu = FakeEcu(pageOverrides = mapOf(0 to DemoPageFactory.page0()))
        val transport = FakeEcuTransport(ecu)
        transport.connect()
        try {
            val reader = EcuMemoryReader(MsProtocolClient(transport), definition)
            val page = reader.readPage(0)
            assertEquals((800..6400 step 800).map { it.toDouble() }, page.decodeArray("veRpmBins"))
            assertEquals((-40..100 step 20).map { it.toDouble() }, page.decodeArray("veCltBins"))
            assertEquals(100.0, page.decodeScalar("demoLimit")!!, 1e-9)
            val ve = page.decodeArray("veTable1Map")!!
            assertEquals(64, ve.size)
            for (r in 0 until 8) for (c in 0 until 8) {
                assertEquals(DemoPageFactory.veValue(r, c), ve[r * 8 + c], 1e-9)
            }
            val ign = page.decodeArray("ignTable1Map")!!
            for (r in 0 until 8) for (c in 0 until 8) {
                assertEquals(DemoPageFactory.ignValue(r, c), ign[r * 8 + c], 1e-9)
            }
        } finally {
            transport.disconnect()
        }
    }

    @Test
    fun `page1 mengikuti pola default untuk test memori`() = runBlocking {
        val ecu = FakeEcu(pageOverrides = mapOf(1 to DemoPageFactory.page1()))
        val transport = FakeEcuTransport(ecu)
        transport.connect()
        try {
            val reader = EcuMemoryReader(MsProtocolClient(transport), definition)
            val page = reader.readPage(1)
            assertEquals(100.0, page.decodeScalar("demoTrim")!!, 1e-9)
        } finally {
            transport.disconnect()
        }
    }
}
