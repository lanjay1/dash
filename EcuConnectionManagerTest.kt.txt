class FakeAppSettings : AppSettings {
    private val _themeMode = kotlinx.coroutines.flow.MutableStateFlow(ThemeMode.SYSTEM)
    private val _baud = kotlinx.coroutines.flow.MutableStateFlow(115200)
    private val _auto = kotlinx.coroutines.flow.MutableStateFlow(true)
    private val _max = kotlinx.coroutines.flow.MutableStateFlow(3)
    private val _delay = kotlinx.coroutines.flow.MutableStateFlow(1_000L)
    private val _scenario = kotlinx.coroutines.flow.MutableStateFlow(DemoScenario.STABLE)
    override val themeMode get() = _themeMode
    override val usbBaudRate get() = _baud
    override val autoReconnect get() = _auto
    override val reconnectMaxAttempts get() = _max
    override val reconnectDelayMs get() = _delay
    override val demoScenario get() = _scenario
    override fun setThemeMode(mode: ThemeMode) { _themeMode.value = mode }
    override fun setUsbBaudRate(baud: Int) { _baud.value = baud }
    override fun setAutoReconnect(enabled: Boolean) { _auto.value = enabled }
    override fun setReconnectMaxAttempts(value: Int) { _max.value = value }
    override fun setReconnectDelayMs(value: Long) { _delay.value = value }
    override fun setDemoScenario(scenario: DemoScenario) { _scenario.value = scenario }
}

/** Transport yang incoming-nya selesai (putus) setelah [dieAfterMs]. */
class DyingTransport(private val dieAfterMs: Long) : EcuTransport {
    override val name = "dying"
    override val incoming: Flow<ByteArray> = flow { delay(dieAfterMs) }
    override suspend fun connect() {}
    override suspend fun disconnect() {}
    override suspend fun write(bytes: ByteArray) {}
}

class FakeSource(transports: List<EcuTransport>) : EcuTransportSource {
    private val queue = ArrayDeque(transports)
    val built = mutableListOf<EcuTarget>()
    override suspend fun build(target: EcuTarget): EcuTransport {
        built += target
        return queue.removeFirstOrNull() ?: error("Antrean transport habis")
    }
    override fun deviceStillPresent(target: EcuTarget): Boolean = true
    override val usbDetachEvents: Flow<android.hardware.usb.UsbDevice> = emptyFlow()
}

class EcuConnectionManagerTest {

    @Test
    fun `koneksi mati lalu tersambung ulang otomatis`() = runTest {
        val settings = FakeAppSettings().apply {
            setReconnectDelayMs(1_000); setReconnectMaxAttempts(3); setAutoReconnect(true)
        }
        val source = FakeSource(listOf(DyingTransport(100), FakeEcuTransport()))
        val manager = EcuConnectionManager(backgroundScope, source, settings)

        manager.connect(EcuTarget.Demo(DemoScenario.STABLE))
        assertEquals(EcuConnectionStatus.CONNECTED, manager.state.value.status)

        advanceTimeBy(3_000)

        assertEquals(EcuConnectionStatus.CONNECTED, manager.state.value.status)
        assertEquals(2, source.built.size)                 // transport kedua dibangun
        assertEquals("Demo ECU (in-process)", manager.state.value.transportName)
    }

    @Test
    fun `reconnect gagal mencapai ERROR setelah percobaan habis`() = runTest {
        val settings = FakeAppSettings().apply {
            setReconnectDelayMs(100); setReconnectMaxAttempts(2); setAutoReconnect(true)
        }
        val source = FakeSource(
            listOf(DyingTransport(50), DyingTransport(50), DyingTransport(50))
        )
        val manager = EcuConnectionManager(backgroundScope, source, settings)

        manager.connect(EcuTarget.Demo(DemoScenario.STABLE))
        advanceTimeBy(5_000)

        assertEquals(EcuConnectionStatus.ERROR, manager.state.value.status)
        assertEquals(2, manager.state.value.reconnectAttempt)
        assertTrue(manager.state.value.lastError!!.contains("menyambung ulang"))
    }

    @Test
    fun `auto reconnect dimatikan langsung ERROR tanpa percobaan`() = runTest {
        val settings = FakeAppSettings().apply { setAutoReconnect(false) }
        val source = FakeSource(listOf(DyingTransport(50)))
        val manager = EcuConnectionManager(backgroundScope, source, settings)

        manager.connect(EcuTarget.Demo(DemoScenario.STABLE))
        advanceTimeBy(1_000)

        assertEquals(EcuConnectionStatus.ERROR, manager.state.value.status)
        assertEquals(1, source.built.size)
        assertTrue(manager.state.value.lastError!!.contains("terputus"))
    }

    @Test
    fun `disconnect manual membatalkan reconnect berjalan`() = runTest {
        val settings = FakeAppSettings().apply {
            setReconnectDelayMs(1_000); setReconnectMaxAttempts(5); setAutoReconnect(true)
        }
        val source = FakeSource(listOf(DyingTransport(100)))
        val manager = EcuConnectionManager(backgroundScope, source, settings)

        manager.connect(EcuTarget.Demo(DemoScenario.STABLE))
        advanceTimeBy(300)                       // mati → RECONNECTING
        assertEquals(EcuConnectionStatus.RECONNECTING, manager.state.value.status)

        manager.disconnect()
        assertEquals(EcuConnectionStatus.DISCONNECTED, manager.state.value.status)

        advanceTimeBy(10_000)                    // loop lama harus batal (generation guard)
        assertEquals(EcuConnectionStatus.DISCONNECTED, manager.state.value.status)
        assertEquals(1, source.built.size)
    }
}
