package com.ztune.app.data.transport

import android.hardware.usb.UsbDeviceConnection
import com.hoho.android.usbserial.driver.UsbSerialPort
import com.ztune.app.domain.protocol.EcuCommunicationException
import com.ztune.app.domain.protocol.EcuTransport
import kotlinx.coroutines.CancellationException
import kotlinx.coroutines.CoroutineScope
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.Job
import kotlinx.coroutines.SupervisorJob
import kotlinx.coroutines.cancel
import kotlinx.coroutines.channels.Channel
import kotlinx.coroutines.flow.Flow
import kotlinx.coroutines.flow.receiveAsFlow
import kotlinx.coroutines.isActive
import kotlinx.coroutines.launch
import kotlinx.coroutines.withContext
import java.io.IOException

/**
 * EcuTransport di atas usb-serial-for-android (CP210x, FTDI, CH340, PL2303, CDC-ACM).
 * Baud rate untuk USB CDC virtual COM diabaikan ECU, tapi tetap diset standar
 * (115200 8N1) karena driver FTDI/CP210x nyata memakainya.
 */
class UsbSerialTransport(
    private val connection: UsbDeviceConnection,
    private val port: UsbSerialPort,
    private val baudRate: Int = DEFAULT_BAUD_RATE,
    override val name: String
) : EcuTransport {

    private val scope = CoroutineScope(SupervisorJob() + Dispatchers.IO)
    private val inbound = Channel<ByteArray>(capacity = 256)
    private var readerJob: Job? = null
    private var closed = false

    @Volatile
    private var portOpen = false

    override val incoming: Flow<ByteArray> = inbound.receiveAsFlow()

    override suspend fun connect() {
        check(!closed) { "Transport sudah ditutup (single-use)" }
        withContext(Dispatchers.IO) {
            try {
                port.open(connection)
                port.setParameters(baudRate, UsbSerialPort.DATABITS_8, UsbSerialPort.STOPBITS_1, UsbSerialPort.PARITY_NONE)
            } catch (e: IOException) {
                throw EcuCommunicationException.ConnectionClosed(e)
            }
        }
        portOpen = true
        readerJob = scope.launch { readLoop() }
    }

    private suspend fun readLoop() {
        val buffer = ByteArray(4096)
        var cause: Throwable? = null
        try {
            while (isActive && portOpen) {
                // Blocking read dengan timeout pendek: 0 = tidak ada data → loop lagi
                // (titik cancel yang responsif). Error/cabut → IOException.
                val n = port.read(buffer, READ_TIMEOUT_MS)
                if (n > 0) inbound.send(buffer.copyOf(n))
            }
        } catch (e: CancellationException) {
            throw e
        } catch (e: Throwable) {
            cause = e
        } finally {
            closeInbound(cause)
        }
    }

    override suspend fun write(bytes: ByteArray) {
        check(portOpen) { "Port USB belum dibuka" }
        withContext(Dispatchers.IO) {
            try {
                port.write(bytes, WRITE_TIMEOUT_MS)
            } catch (e: IOException) {
                throw EcuCommunicationException.ConnectionClosed(e)
            }
        }
    }

    override suspend fun disconnect() {
        if (closed) return
        closed = true
        portOpen = false
        readerJob?.cancel()
        readerJob = null
        withContext(Dispatchers.IO) {
            try {
                port.close() // juga menutup UsbDeviceConnection
            } catch (_: Exception) {
                // abaikan — kita sedang menutup apa pun yang terjadi
            }
        }
        closeInbound()
        scope.cancel()
    }

    private fun closeInbound(cause: Throwable? = null) {
        if (!inbound.isClosedForSend) {
            if (cause != null) inbound.close(cause) else inbound.close()
        }
    }

    companion object {
        const val DEFAULT_BAUD_RATE = 115200
        const val READ_TIMEOUT_MS = 150
        const val WRITE_TIMEOUT_MS = 1_000
    }
}
