package com.ztune.app.presentation.dashboard

import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import com.ztune.app.data.definition.EcuDefinitionRepository
import com.ztune.app.data.transport.EcuConnectionManager
import com.ztune.app.data.transport.EcuConnectionStatus
import com.ztune.app.domain.ini.model.ConstantDefinition
import com.ztune.app.domain.ini.model.ConstantKind
import com.ztune.app.domain.memory.LiveDataFrame
import com.ztune.app.domain.memory.LiveDataStreamer
import dagger.hilt.android.lifecycle.HiltViewModel
import javax.inject.Inject
import kotlinx.coroutines.CancellationException
import kotlinx.coroutines.Job
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch

data class GaugeDisplay(
    val channel: String,
    val units: String,
    val min: Double,
    val max: Double,
    val digits: Int,
    val value: Double? = null
)

data class DashboardUiState(
    val connectionStatus: EcuConnectionStatus = EcuConnectionStatus.DISCONNECTED,
    val definitionLoaded: Boolean = false,
    val parseIssueCount: Int = 0,
    val isStreaming: Boolean = false,
    val primaryGauges: List<GaugeDisplay> = emptyList(),
    val secondaryGauges: List<GaugeDisplay> = emptyList(),
    val sampleCount: Long = 0L,
    val rateHz: Double = 0.0,
    val error: String? = null
)

@HiltViewModel
class DashboardViewModel @Inject constructor(
    private val connectionManager: EcuConnectionManager,
    private val definitionRepository: EcuDefinitionRepository
) : ViewModel() {

    private val _uiState = MutableStateFlow(DashboardUiState())
    val uiState: StateFlow<DashboardUiState> = _uiState.asStateFlow()

    private var streamJob: Job? = null
    private var lastFrameTimestamp = 0L
    private var rateEma = 0.0

    init {
        definitionRepository.ensureBundledLoaded()
        buildGaugeSkeleton()
        viewModelScope.launch {
            connectionManager.state.collect { state ->
                _uiState.update { it.copy(connectionStatus = state.status) }
                if (state.status != EcuConnectionStatus.CONNECTED && streamJob?.isActive == true) {
                    stopStreaming()
                }
            }
        }
    }

    fun startStreaming() {
        if (streamJob?.isActive == true) return
        if (definitionRepository.definition.value == null) {
            _uiState.update { it.copy(error = "Definisi ECU belum dimuat") }
            return
        }
        if (!connectionManager.isConnected) {
            _uiState.update { it.copy(error = "Belum terhubung ke ECU") }
            return
        }
        streamJob = viewModelScope.launch {
            _uiState.update {
                it.copy(isStreaming = true, error = null, sampleCount = 0L, rateHz = 0.0)
            }
            lastFrameTimestamp = 0L
            rateEma = 0.0
            try {
                val def = definitionRepository.definition.value
                    ?: error("Definisi ECU belum dimuat")
                connectionManager.withClient { client ->
                    val streamer = LiveDataStreamer(client, def)
                    streamer.stream(periodMs = POLL_PERIOD_MS).collect { frame ->
                        onFrame(frame)
                    }
                }
            } catch (e: CancellationException) {
                throw e
            } catch (e: Exception) {
                _uiState.update { it.copy(error = e.message ?: e.javaClass.simpleName) }
            } finally {
                _uiState.update { it.copy(isStreaming = false) }
            }
        }
    }

    fun stopStreaming() {
        streamJob?.cancel()
        streamJob = null
    }

    private fun onFrame(frame: LiveDataFrame) {
        val dt = frame.timestampMs - lastFrameTimestamp
        if (dt in 1..5000) {
            val instant = 1000.0 / dt
            rateEma = if (rateEma <= 0.0) instant else rateEma * 0.8 + instant * 0.2
        }
        lastFrameTimestamp = frame.timestampMs
        _uiState.update { s ->
            s.copy(
                sampleCount = s.sampleCount + 1,
                rateHz = rateEma,
                primaryGauges = s.primaryGauges.map { g -> g.copy(value = frame.values[g.channel]) },
                secondaryGauges = s.secondaryGauges.map { g -> g.copy(value = frame.values[g.channel]) }
            )
        }
    }

    private fun buildGaugeSkeleton() {
        val def = definitionRepository.definition.value ?: return
        val allScalars = def.outputChannels.values.filter { it.kind == ConstantKind.SCALAR }
        val primaries = primaryNames.mapNotNull { name ->
            allScalars.firstOrNull { it.name.equals(name, ignoreCase = true) }
        }.map { it.toDisplay() }
        val secondaries = allScalars
            .filter { c -> primaries.none { it.channel.equals(c.name, ignoreCase = true) } }
            .sortedBy { it.offset }
            .map { it.toDisplay() }
        _uiState.update {
            it.copy(
                primaryGauges = primaries,
                secondaryGauges = secondaries,
                definitionLoaded = true,
                parseIssueCount = def.issues.size
            )
        }
    }

    private fun ConstantDefinition.toDisplay(): GaugeDisplay {
        val (min, max) = displayRange(this)
        return GaugeDisplay(
            channel = name,
            units = units,
            min = min,
            max = max,
            digits = digits
        )
    }

    private fun displayRange(c: ConstantDefinition): Pair<Double, Double> {
        displayRangeOverride[c.name.uppercase()]?.let { return it }
        return if (c.max > c.min) c.min to c.max else 0.0 to 100.0
    }

    companion object {
        private val primaryNames = listOf("RPM", "AFR", "TPS", "CLT")
        private val displayRangeOverride = mapOf(
            "RPM" to (0.0 to 8000.0),
            "CLT" to (-40.0 to 120.0),
            "MAP" to (0.0 to 255.0)
        )
        const val POLL_PERIOD_MS = 50L
    }
}
