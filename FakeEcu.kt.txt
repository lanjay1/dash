package com.ztune.app.domain.protocol

/**
 * ECU simulasi (pure Kotlin): 'Q'/'S'/'R'/'O'/'C'/'B'/'k' — tanpa envelope.
 * [pageOverrides] mengganti isi page tertentu (mode demo memakai DemoPageFactory);
 * page tanpa override memakai pola deterministik (dipakai test memori).
 */
class FakeEcu(
    val signature: String = DEFAULT_SIGNATURE,
    pageSize: Int = DEFAULT_PAGE_SIZE,
    val blockingFactor: Int = DEFAULT_BLOCKING_FACTOR,
    val timeSource: () -> Long = System::currentTimeMillis,
    val ochBlockSize: Int = 0,
    val outputBlockProvider: ((Long) -> ByteArray)? = null,
    pageOverrides: Map<Int, ByteArray> = emptyMap()
) {

    val pages: MutableMap<Int, ByteArray> = mutableMapOf(
        0 to (pageOverrides[0] ?: ByteArray(pageSize) { ((it * 7) + 3).toByte() }),
        1 to (pageOverrides[1] ?: ByteArray(pageSize) { ((it * 13) + 100).toByte() })
    )

    /** Simulasi flash: hasil burn terakhir per page (untuk assert test). */
    val committedPages: MutableMap<Int, ByteArray> = pages.mapValues { it.value.copyOf() }.toMutableMap()

    /** Log payload request yang diterima — untuk assert di test. */
    val receivedRequests = mutableListOf<ByteArray>()

    /** Proses SATU paket request ter-envelop → paket respons ter-envelop. */
    fun respondToPacket(packet: ByteArray): ByteArray {
        val payload = decodeRequest(packet)
        val responsePayload = try {
            handlePayload(payload)
        } catch (e: FakeEcuException) {
            return MsEnvelopeCodec.encodeResponse(e.flag, ByteArray(0))
        }
        return MsEnvelopeCodec.encodeResponse(MsEnvelopeCodec.FLAG_OK, responsePayload)
    }

    private fun handlePayload(payload: ByteArray): ByteArray {
        receivedRequests += payload
        if (payload.isEmpty()) throw FakeEcuException(FLAG_UNRECOGNIZED)
        return when (payload[0].toInt().toChar()) {
            'Q', 'S' -> (signature + "\u0000").toByteArray(Charsets.US_ASCII)
            'R' -> {
                if (payload.size < 7) throw FakeEcuException(FLAG_UNDERRUN)
                val page = MsEnvelopeCodec.readU16BE(payload, 1)
                val offset = MsEnvelopeCodec.readU16BE(payload, 3)
                val count = MsEnvelopeCodec.readU16BE(payload, 5)
                val memory = pages[page] ?: throw FakeEcuException(FLAG_OUT_OF_RANGE)
                if (offset + count > memory.size) throw FakeEcuException(FLAG_OUT_OF_RANGE)
                memory.copyOfRange(offset, offset + count)
            }
            'O' -> {
                if (payload.size < 5) throw FakeEcuException(FLAG_UNDERRUN)
                val provider = outputBlockProvider ?: throw FakeEcuException(FLAG_UNRECOGNIZED)
                if (ochBlockSize <= 0) throw FakeEcuException(FLAG_UNRECOGNIZED)
                val offset = MsEnvelopeCodec.readU16BE(payload, 1)
                val count = MsEnvelopeCodec.readU16BE(payload, 3)
                if (offset + count > ochBlockSize) throw FakeEcuException(FLAG_OUT_OF_RANGE)
                val block = provider(timeSource())
                if (block.size < ochBlockSize) throw FakeEcuException(FLAG_OUT_OF_RANGE)
                block.copyOfRange(offset, offset + count)
            }
            'C' -> {
                if (payload.size < 7) throw FakeEcuException(FLAG_UNDERRUN)
                val page = MsEnvelopeCodec.readU16BE(payload, 1)
                val offset = MsEnvelopeCodec.readU16BE(payload, 3)
                val count = MsEnvelopeCodec.readU16BE(payload, 5)
                if (payload.size < 7 + count) throw FakeEcuException(FLAG_UNDERRUN)
                val memory = pages[page] ?: throw FakeEcuException(FLAG_OUT_OF_RANGE)
                if (offset + count > memory.size) throw FakeEcuException(FLAG_OUT_OF_RANGE)
                payload.copyInto(memory, offset, 7, 7 + count)
                ByteArray(0)
            }
            'B' -> {
                if (payload.size < 3) throw FakeEcuException(FLAG_UNDERRUN)
                val page = MsEnvelopeCodec.readU16BE(payload, 1)
                val memory = pages[page] ?: throw FakeEcuException(FLAG_OUT_OF_RANGE)
                committedPages[page] = memory.copyOf()
                ByteArray(0)
            }
            'k' -> {
                if (payload.size < 7) throw FakeEcuException(FLAG_UNDERRUN)
                val page = MsEnvelopeCodec.readU16BE(payload, 1)
                val offset = MsEnvelopeCodec.readU16BE(payload, 3)
                val count = MsEnvelopeCodec.readU16BE(payload, 5)
                val memory = pages[page] ?: throw FakeEcuException(FLAG_OUT_OF_RANGE)
                if (offset + count > memory.size) throw FakeEcuException(FLAG_OUT_OF_RANGE)
                val crc = Crc32.compute(memory, offset, count)
                val out = ByteArray(4)
                out[0] = (crc and 0xFF).toByte()
                out[1] = ((crc shr 8) and 0xFF).toByte()
                out[2] = ((crc shr 16) and 0xFF).toByte()
                out[3] = ((crc shr 24) and 0xFF).toByte()
                out
            }
            else -> throw FakeEcuException(FLAG_UNRECOGNIZED)
        }
    }

    private fun decodeRequest(packet: ByteArray): ByteArray {
        if (packet.size < 6) throw IllegalArgumentException("Paket request terlalu pendek: ${packet.size}")
        val size = MsEnvelopeCodec.readU16BE(packet, 0)
        if (size != packet.size) {
            throw IllegalArgumentException("Size header $size != panjang paket ${packet.size}")
        }
        val payloadLen = size - 6
        val payload = packet.copyOfRange(2, 2 + payloadLen)
        val wireCrc = MsEnvelopeCodec.readCrc(packet, 2 + payloadLen)
        val calcCrc = Crc32.compute(payload).toLong() and 0xFFFFFFFFL
        if (wireCrc != calcCrc) throw IllegalArgumentException("CRC request tidak cocok")
        return payload
    }

    class FakeEcuException(val flag: Int) : Exception("flag=0x%02X".format(flag))

    companion object {
        const val FLAG_UNDERRUN = 0x80
        const val FLAG_UNRECOGNIZED = 0x83
        const val FLAG_OUT_OF_RANGE = 0x84
        const val DEFAULT_SIGNATURE = "ZTUNE FakeECU (rusEFI MS-envelope compat) v0.1"
        const val DEFAULT_PAGE_SIZE = 256
        const val DEFAULT_BLOCKING_FACTOR = 1024
    }
}
