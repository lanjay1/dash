package com.ztune.app.data.transport

import com.ztune.app.data.di.ApplicationScope
import com.ztune.app.data.settings.AppSettings
import com.ztune.app.domain.protocol.EcuTransport
import com.ztune.app.domain.protocol.MsProtocolClient
import javax.inject.Inject
import javax.inject.Singleton
import kotlinx.coroutines.CoroutineScope
import kotlinx.coroutines.Job
import kotlinx.coroutines.delay
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch

enum class EcuConnectionStatus { DISCONNECTED, CONNECTING, CONNECTED, RECONNECTING, ERROR }

data class EcuConnectionState(
    val status: EcuConnectionStatus = EcuConnectionStatus.DISCONNECTED,
    val transportName: String? = null,
    val signature: String? = null,
    val lastError: String? = null,
    val reconnectAttempt: Int = 0,
    val reconnectMaxAttempts: Int = 0
)

/**
 * Pemilik status koneksi ECU + logika auto-reconnect.
 *
 * - connect(target): bangun transport via [EcuTransportSource], lalu pasang
 *   watcher pada incoming-flow; selesai/error-nya stream = koneksi mati.
 * - markLost: auto-reconnect (jika settings aktif) → RECONNECTING → loop
 *   build+connect hingga sukses atau maxAttempts habis → ERROR.
 * - Generation guard: setiap disconnect/connect menaikkan [generation];
 *   loop/watcher lama otomatis batal saat guard tidak cocok.
 * - Timeout operasi protokol TIDAK memicu reconnect (ECU sibuk ≠ kabel cabut).
 * - Reconnect sukses: signature dikosongkan (handshake ulang manual opsional).
 * - Bebas Android: semua I/O sistem lewat [EcuTransportSource] → testable.
 */
@Singleton
class EcuConnectionManager @Inject constructor(
    @ApplicationScope private val scope: CoroutineScope,
    private val source: EcuTransportSource,
    private val settings: AppSettings
) {

    private val _state = MutableStateFlow(EcuConnectionState())
    val state: StateFlow<EcuConnectionState> = _state.asStateFlow()

    private var transport: EcuTransport? = null
    private var currentTarget: EcuTarget? = null
    private var watchJob: Job? = null
    private var generation = 0L

    val isConnected: Boolean get() = transport != null

    init {
        scope.launch {
            source.usbDetachEvents.collect { device ->
                val target = currentTarget
                if (device != null && target is EcuTarget.Usb &&
                    target.device.deviceId == device.deviceId &&
                    transport != null &&
                    _state.value.status == EcuConnectionStatus.CONNECTED
                ) {
                    markLost("Device USB dilepas")
                }
            }
        }
    }

    suspend fun connect(target: EcuTarget) {
        disconnectInternal()
        val gen = ++generation
        currentTarget = target
        _state.update {
            it.copy(
                status = EcuConnectionStatus.CONNECTING,
                transportName = null,
                signature = null,
                lastError = null,
                reconnectAttempt = 0,
                reconnectMaxAttempts = 0
            )
        }
        try {
            val t = source.build(target)
            t.connect()
            if (gen != generation) {
                runCatching { t.disconnect() }
                return
            }
            transport = t
            _state.update { it.copy(status = EcuConnectionStatus.CONNECTED, transportName = t.name) }
            startWatcher(t, gen)
        } catch (e: Exception) {
            transport = null
            _state.update {
                it.copy(status = EcuConnectionStatus.ERROR, lastError = e.message ?: e.javaClass.simpleName)
            }
            throw e
        }
    }

    suspend fun disconnect() {
        disconnectInternal()
        currentTarget = null
        _state.update {
            it.copy(
                status = EcuConnectionStatus.DISCONNECTED,
                transportName = null,
                signature = null,
                lastError = null,
                reconnectAttempt = 0,
                reconnectMaxAttempts = 0
            )
        }
    }

    suspend fun handshake(commandTemplate: String = "Q"): String {
        val signature = withClient { it.querySignature(commandTemplate) }
        _state.update { it.copy(signature = signature) }
        return signature
    }

    suspend fun <T> withClient(block: suspend (MsProtocolClient) -> T): T {
        val current = transport ?: error("Belum terhubung ke ECU — connect() dulu")
        return block(MsProtocolClient(current))
    }

    // ---------- internal ----------

    private fun startWatcher(t: EcuTransport, gen: Long) {
        watchJob = scope.launch {
            try {
                t.incoming.collect { }
            } catch (e: CancellationException) {
                throw e
            } catch (_: Throwable) {
                // sebab dicatat lewat markLost
            }
            if (gen == generation && transport === t) {
                markLost("Koneksi ke ECU terputus")
            }
        }
    }

    private suspend fun markLost(reason: String?) {
        val target = currentTarget ?: return
        watchJob?.cancel(); watchJob = null
        transport?.let { runCatching { it.disconnect() } }
        transport = null
        if (!settings.autoReconnect.value) {
            _state.update {
                it.copy(
                    status = EcuConnectionStatus.ERROR,
                    lastError = reason ?: "Koneksi terputus",
                    reconnectAttempt = 0,
                    reconnectMaxAttempts = 0
                )
            }
            return
        }
        val gen = generation
        scope.launch { reconnectLoop(gen, target, reason) }
    }

    private suspend fun reconnectLoop(gen: Long, target: EcuTarget, reason: String?) {
        val max = settings.reconnectMaxAttempts.value.coerceAtLeast(1)
        val delayMs = settings.reconnectDelayMs.value.coerceAtLeast(100)
        _state.update {
            it.copy(
                status = EcuConnectionStatus.RECONNECTING,
                reconnectMaxAttempts = max,
                reconnectAttempt = 0,
                lastError = reason
            )
        }
        var lastError: String? = reason
        for (attempt in 1..max) {
            if (gen != generation) return
            _state.update { it.copy(reconnectAttempt = attempt) }
            delay(delayMs)
            if (gen != generation) return
            try {
                val t = source.build(target)
                t.connect()
                if (gen != generation) {
                    runCatching { t.disconnect() }
                    return
                }
                transport = t
                _state.update {
                    it.copy(
                        status = EcuConnectionStatus.CONNECTED,
                        transportName = t.name,
                        signature = null,
                        lastError = null
                    )
                }
                startWatcher(t, gen)
                return
            } catch (e: Exception) {
                lastError = e.message ?: e.javaClass.simpleName
            }
        }
        if (gen == generation) {
            _state.update {
                it.copy(
                    status = EcuConnectionStatus.ERROR,
                    lastError = "Koneksi terputus — gagal menyambung ulang setelah $max percobaan" +
                        (lastError?.let { e -> " ($e)" } ?: "")
                )
            }
        }
    }

    private suspend fun disconnectInternal() {
        generation++
        watchJob?.cancel(); watchJob = null
        transport?.let { runCatching { it.disconnect() } }
        transport = null
    }
}
