package com.ztune.app.domain.ini.model

enum class Endianness { LITTLE, BIG }

data class EcuMeta(
    val signature: String? = null,
    val endianness: Endianness? = null
)

/**
 * Template command serial mentah persis seperti tertulis di INI
 * (mis. "R%2i%2o%2c"). Interpretasi placeholder %2i/%2o/%2c/%v
 * dilakukan ProtocolProfile di FASE 2/3 — parser hanya meneruskan.
 */
data class EcuCommands(
    val queryCommand: String? = null,
    val versionInfo: String? = null,
    val pageReadCommand: String? = null,
    val pageChunkWrite: String? = null,
    val burnCommand: String? = null,
    val crc32CheckCommand: String? = null,
    val pageActivationCommand: String? = null,
    val ochGetCommand: String? = null,
    val ochBlockSize: Int? = null,
    val blockingFactor: Int? = null
)

/** Assignment biasa yang bukan definisi (mis. nilai default / ekspresi) — disimpan mentah. */
data class ExtraEntry(
    val section: String,
    val key: String,
    val value: String,
    val lineNumber: Int
)

/**
 * Hasil akhir parsing satu file .ini definisi ECU.
 * Immutable; konsumen (repository FASE 4) tinggal menyimpan/meneruskannya.
 */
data class EcuDefinition(
    val meta: EcuMeta = EcuMeta(),
    val commands: EcuCommands = EcuCommands(),
    val pages: List<PageDefinition> = emptyList(),
    val constants: Map<String, ConstantDefinition> = emptyMap(),
    val tables: Map<String, TableDefinition> = emptyMap(),
    val outputChannels: Map<String, ConstantDefinition> = emptyMap(),
    val extraEntries: List<ExtraEntry> = emptyList(),
    val issues: List<IniParseIssue> = emptyList()
)
