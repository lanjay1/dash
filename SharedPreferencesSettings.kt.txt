package com.ztune.app.data.settings

import android.content.Context
import androidx.core.content.edit
import com.ztune.app.data.transport.DemoScenario
import dagger.hilt.android.qualifiers.ApplicationContext
import javax.inject.Inject
import javax.inject.Singleton
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update

@Singleton
class SharedPreferencesSettings @Inject constructor(
    @ApplicationContext context: Context
) : AppSettings {

    private val prefs = context.getSharedPreferences(PREFS_NAME, Context.MODE_PRIVATE)

    private fun <T> read(key: String, fallback: T, convert: (String) -> T?): T =
        prefs.getString(key, null)?.let { runCatching { convert(it) }.getOrNull() } ?: fallback

    private val _themeMode = MutableStateFlow(
        read(K_THEME, ThemeMode.SYSTEM) { ThemeMode.valueOf(it) }
    )
    private val _usbBaudRate = MutableStateFlow(
        read(K_BAUD, DEFAULT_BAUD) { it.toInt() }
    )
    private val _autoReconnect = MutableStateFlow(prefs.getBoolean(K_AUTO_RECONNECT, true))
    private val _maxAttempts = MutableStateFlow(
        read(K_MAX_ATTEMPTS, DEFAULT_MAX_ATTEMPTS) { it.toInt() }
    )
    private val _delayMs = MutableStateFlow(
        read(K_DELAY_MS, DEFAULT_DELAY_MS) { it.toLong() }
    )
    private val _demoScenario = MutableStateFlow(
        read(K_DEMO_SCENARIO, DemoScenario.STABLE) { DemoScenario.valueOf(it) }
    )

    override val themeMode: StateFlow<ThemeMode> = _themeMode.asStateFlow()
    override val usbBaudRate: StateFlow<Int> = _usbBaudRate.asStateFlow()
    override val autoReconnect: StateFlow<Boolean> = _autoReconnect.asStateFlow()
    override val reconnectMaxAttempts: StateFlow<Int> = _maxAttempts.asStateFlow()
    override val reconnectDelayMs: StateFlow<Long> = _delayMs.asStateFlow()
    override val demoScenario: StateFlow<DemoScenario> = _demoScenario.asStateFlow()

    override fun setThemeMode(mode: ThemeMode) {
        prefs.edit { putString(K_THEME, mode.name) }
        _themeMode.update { mode }
    }

    override fun setUsbBaudRate(baud: Int) {
        prefs.edit { putString(K_BAUD, baud.toString()) }
        _usbBaudRate.update { baud }
    }

    override fun setAutoReconnect(enabled: Boolean) {
        prefs.edit { putBoolean(K_AUTO_RECONNECT, enabled) }
        _autoReconnect.update { enabled }
    }

    override fun setReconnectMaxAttempts(value: Int) {
        prefs.edit { putString(K_MAX_ATTEMPTS, value.toString()) }
        _maxAttempts.update { value }
    }

    override fun setReconnectDelayMs(value: Long) {
        prefs.edit { putString(K_DELAY_MS, value.toString()) }
        _delayMs.update { value }
    }

    override fun setDemoScenario(scenario: DemoScenario) {
        prefs.edit { putString(K_DEMO_SCENARIO, scenario.name) }
        _demoScenario.update { scenario }
    }

    companion object {
        private const val PREFS_NAME = "ztune_settings"
        private const val K_THEME = "theme_mode"
        private const val K_BAUD = "usb_baud_rate"
        private const val K_AUTO_RECONNECT = "auto_reconnect"
        private const val K_MAX_ATTEMPTS = "reconnect_max_attempts"
        private const val K_DELAY_MS = "reconnect_delay_ms"
        private const val K_DEMO_SCENARIO = "demo_scenario"
        const val DEFAULT_BAUD = 115200
        const val DEFAULT_MAX_ATTEMPTS = 5
        const val DEFAULT_DELAY_MS = 2_000L
    }
}
