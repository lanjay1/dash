package com.ztune.app.domain.table

import com.ztune.app.domain.ini.BundledEcuDefinition
import com.ztune.app.domain.ini.IniDefinitionParser
import com.ztune.app.domain.ini.model.Endianness
import org.junit.Assert.assertEquals
import org.junit.Assert.assertNull
import org.junit.Test

class TableInterpolatorTest {

    private fun grid(): TableGrid {
        val definition = IniDefinitionParser().parse(BundledEcuDefinition.INI)
        val table = definition.tables.getValue("veTable1Tbl")
        val map = definition.constants.getValue(table.mapName!!)
        return TableGrid(table, map, ByteArray(256), Endianness.LITTLE)
    }

    @Test
    fun `row wise linear antar dua anchor`() {
        val g = grid()
        g.setPhysical(0, 0, 10.0)
        g.setPhysical(0, 4, 30.0)
        val batch = TableInterpolator.rowWise(g, setOf(CellPos(0, 0), CellPos(0, 4)))!!
        assertEquals(3, batch.edits.size)
        g.applyForTest(batch)
        assertEquals(10.0, g.physical(0, 0)!!, 1e-9)
        assertEquals(15.0, g.physical(0, 1)!!, 1e-9)
        assertEquals(20.0, g.physical(0, 2)!!, 1e-9)
        assertEquals(25.0, g.physical(0, 3)!!, 1e-9)
        assertEquals(30.0, g.physical(0, 4)!!, 1e-9)
    }

    @Test
    fun `column wise bekerja per kolom`() {
        val g = grid()
        g.setPhysical(1, 3, 0.0)
        g.setPhysical(5, 3, 40.0)
        val batch = TableInterpolator.columnWise(g, setOf(CellPos(1, 3), CellPos(5, 3)))!!
        assertEquals(3, batch.edits.size)
        g.applyForTest(batch)
        assertEquals(10.0, g.physical(2, 3)!!, 1e-9)
        assertEquals(30.0, g.physical(4, 3)!!, 1e-9)
    }

    @Test
    fun `sel lain di baris sama tidak tersentuh`() {
        val g = grid()
        g.setPhysical(0, 0, 10.0)
        g.setPhysical(0, 4, 30.0)
        g.setPhysical(0, 7, 99.0)
        g.applyForTest(TableInterpolator.rowWise(g, setOf(CellPos(0, 0), CellPos(0, 4)))!!)
        assertEquals(99.0, g.physical(0, 7)!!, 1e-9)
    }

    @Test
    fun `satu sel terpilih mengembalikan null`() {
        assertNull(TableInterpolator.rowWise(grid(), setOf(CellPos(0, 0))))
    }

    private fun TableGrid.applyForTest(batch: EditBatch) {
        batch.edits.forEach { setPhysical(it.row, it.col, it.newValue) }
    }
}
