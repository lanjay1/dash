package com.ztune.app.domain.protocol

/**
 * CRC-32 standar (reflected, polinomial 0xEDB88320, init 0xFFFFFFFF, final XOR).
 * Identik dengan java.util.zip.CRC32 dan crc32() di tunerstudio_io.cpp rusEFI.
 * Implementasi sendiri agar domain tetap pure-Kotlin dan vektor uji eksplisit.
 */
object Crc32 {

    private const val POLY = 0xEDB88320u.toInt()

    private val TABLE = IntArray(256) { index ->
        var crc = index
        repeat(8) {
            crc = if (crc and 1 != 0) (crc ushr 1) xor POLY else crc ushr 1
        }
        crc
    }

    fun compute(data: ByteArray, offset: Int = 0, length: Int = data.size - offset): Int {
        require(offset >= 0 && length >= 0 && offset + length <= data.size) {
            "CRC range tidak valid: offset=$offset length=$length size=${data.size}"
        }
        var crc = -1
        for (i in offset until offset + length) {
            crc = (crc ushr 8) xor TABLE[(crc xor data[i].toInt()) and 0xFF]
        }
        return crc.inv()
    }
}
