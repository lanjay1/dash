@Composable
private fun statusLabel(status: EcuConnectionStatus): String = when (status) {
    EcuConnectionStatus.DISCONNECTED -> stringResource(R.string.conn_status_disconnected)
    EcuConnectionStatus.CONNECTING -> stringResource(R.string.conn_status_connecting)
    EcuConnectionStatus.CONNECTED -> stringResource(R.string.conn_status_connected)
    EcuConnectionStatus.RECONNECTING -> stringResource(R.string.conn_status_reconnecting)
    EcuConnectionStatus.ERROR -> stringResource(R.string.conn_status_error)
}

@Composable
fun ConnectionScreen(
    modifier: Modifier = Modifier,
    viewModel: ConnectionViewModel = hiltViewModel()
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    ConnectionContent(
        uiState = uiState,
        onRefresh = viewModel::refreshUsbDevices,
        onSelectDevice = viewModel::selectDevice,
        onConnectUsb = viewModel::connectUsb,
        onConnectDemo = viewModel::connectDemo,
        onHandshakeQ = { viewModel.checkSignature("Q") },
        onHandshakeS = { viewModel.checkSignature("S") },
        onReadPage = viewModel::readPagePreview,
        onDisconnect = viewModel::disconnect,
        modifier = modifier
    )
}

@Composable
private fun ConnectionContent(
    uiState: ConnectionUiState,
    onRefresh: () -> Unit,
    onSelectDevice: (String) -> Unit,
    onConnectUsb: () -> Unit,
    onConnectDemo: () -> Unit,
    onHandshakeQ: () -> Unit,
    onHandshakeS: () -> Unit,
    onReadPage: () -> Unit,
    onDisconnect: () -> Unit,
    modifier: Modifier = Modifier
) {
    val connected = uiState.connection.status == EcuConnectionStatus.CONNECTED
    val reconnecting = uiState.connection.status == EcuConnectionStatus.RECONNECTING
    Column(
        modifier = modifier
            .fillMaxSize()
            .verticalScroll(rememberScrollState())
            .padding(16.dp),
        verticalArrangement = Arrangement.spacedBy(12.dp)
    ) {
        Text(stringResource(R.string.conn_title), style = MaterialTheme.typography.headlineMedium)
        Text(
            stringResource(R.string.conn_subtitle),
            style = MaterialTheme.typography.bodyMedium,
            color = MaterialTheme.colorScheme.onSurfaceVariant
        )

        StatusCard(uiState.connection)

        Text(stringResource(R.string.conn_usb_section), style = MaterialTheme.typography.titleLarge)
        OutlinedButton(onClick = onRefresh, enabled = !uiState.busy) {
            Text(stringResource(R.string.conn_refresh))
        }
        if (uiState.usbDevices.isEmpty()) {
            Text(
                stringResource(R.string.conn_no_devices),
                style = MaterialTheme.typography.bodyMedium,
                color = MaterialTheme.colorScheme.onSurfaceVariant
            )
        } else {
            uiState.usbDevices.forEach { device ->
                DeviceRow(
                    description = device.description,
                    selected = device.deviceKey == uiState.selectedDeviceKey,
                    enabled = !uiState.busy,
                    onSelect = { onSelectDevice(device.deviceKey) }
                )
            }
            Button(
                onClick = onConnectUsb,
                enabled = !uiState.busy && !connected && !reconnecting && uiState.selectedDeviceKey != null,
                modifier = Modifier.fillMaxWidth()
            ) {
                Text(stringResource(R.string.conn_connect_usb))
            }
        }

        HorizontalDivider()

        Text(stringResource(R.string.conn_demo_section), style = MaterialTheme.typography.titleLarge)
        OutlinedButton(
            onClick = onConnectDemo,
            enabled = !uiState.busy && !connected && !reconnecting
        ) {
            Text(stringResource(R.string.conn_demo_button))
        }

        if (connected || reconnecting) {
            HorizontalDivider()
            Text(stringResource(R.string.conn_ops_section), style = MaterialTheme.typography.titleLarge)
            Row(horizontalArrangement = Arrangement.spacedBy(12.dp)) {
                Button(onClick = onHandshakeQ, enabled = !uiState.busy && connected, modifier = Modifier.weight(1f)) {
                    Text(stringResource(R.string.conn_handshake_q))
                }
                Button(onClick = onHandshakeS, enabled = !uiState.busy && connected, modifier = Modifier.weight(1f)) {
                    Text(stringResource(R.string.conn_handshake_s))
                }
            }
            Button(
                onClick = onReadPage,
                enabled = !uiState.busy && connected,
                modifier = Modifier.fillMaxWidth()
            ) {
                Text(stringResource(R.string.conn_read_page))
            }
            OutlinedButton(
                onClick = onDisconnect,
                enabled = !uiState.busy,
                modifier = Modifier.fillMaxWidth()
            ) {
                Text(stringResource(R.string.conn_disconnect))
            }
        }

        if (uiState.pagePreview != null) {
            Text(stringResource(R.string.conn_preview_title), style = MaterialTheme.typography.titleLarge)
            Text(
                text = uiState.pagePreview!!,
                style = MaterialTheme.typography.bodySmall.copy(fontFamily = FontFamily.Monospace),
                softWrap = false,
                modifier = Modifier.horizontalScroll(rememberScrollState())
            )
        }

        Text(stringResource(R.string.conn_log_title), style = MaterialTheme.typography.titleLarge)
        LazyColumn(
            modifier = Modifier
                .fillMaxWidth()
                .heightIn(max = 220.dp)
        ) {
            items(uiState.log) { line ->
                Text(
                    text = line,
                    style = MaterialTheme.typography.bodySmall.copy(fontFamily = FontFamily.Monospace)
                )
            }
        }
    }
}

@Composable
private fun StatusCard(state: EcuConnectionState, modifier: Modifier = Modifier) {
    val container = when (state.status) {
        EcuConnectionStatus.CONNECTED -> MaterialTheme.colorScheme.primaryContainer
        EcuConnectionStatus.ERROR -> MaterialTheme.colorScheme.errorContainer
        else -> MaterialTheme.colorScheme.surfaceVariant
    }
    val onContainer = when (state.status) {
        EcuConnectionStatus.CONNECTED -> MaterialTheme.colorScheme.onPrimaryContainer
        EcuConnectionStatus.ERROR -> MaterialTheme.colorScheme.onErrorContainer
        else -> MaterialTheme.colorScheme.onSurfaceVariant
    }
    Card(
        modifier = modifier.fillMaxWidth(),
        colors = CardDefaults.cardColors(containerColor = container)
    ) {
        Column(
            modifier = Modifier.padding(16.dp),
            verticalArrangement = Arrangement.spacedBy(4.dp)
        ) {
            Text(statusLabel(state.status), style = MaterialTheme.typography.titleLarge, color = onContainer)
            state.transportName?.let {
                Text("Transport: $it", style = MaterialTheme.typography.bodyMedium, color = onContainer)
            }
            if (state.status == EcuConnectionStatus.RECONNECTING && state.reconnectMaxAttempts > 0) {
                Text(
                    stringResource(R.string.conn_reconnect_fmt, state.reconnectAttempt, state.reconnectMaxAttempts),
                    style = MaterialTheme.typography.bodyMedium.copy(fontFamily = FontFamily.Monospace),
                    color = onContainer
                )
            }
            state.signature?.let {
                Text(
                    "Signature: $it",
                    style = MaterialTheme.typography.bodyMedium.copy(fontFamily = FontFamily.Monospace),
                    color = onContainer
                )
            }
            state.lastError?.let {
                Text("Info: $it", style = MaterialTheme.typography.bodyMedium, color = onContainer)
            }
        }
    }
}

@Composable
private fun DeviceRow(
    description: String,
    selected: Boolean,
    enabled: Boolean,
    onSelect: () -> Unit
) {
    Row(
        modifier = Modifier
            .fillMaxWidth()
            .clickable(enabled = enabled, onClick = onSelect)
            .padding(vertical = 4.dp),
        verticalAlignment = Alignment.CenterVertically
    ) {
        RadioButton(selected = selected, onClick = onSelect, enabled = enabled)
        Text(description, style = MaterialTheme.typography.bodyMedium)
    }
}
