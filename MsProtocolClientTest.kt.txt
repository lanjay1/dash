package com.ztune.app.domain.protocol

import kotlinx.coroutines.awaitCancellation
import kotlinx.coroutines.flow.Flow
import kotlinx.coroutines.flow.flow
import kotlinx.coroutines.runBlocking
import org.junit.Assert.assertArrayEquals
import org.junit.Assert.assertEquals
import org.junit.Assert.assertTrue
import org.junit.Assert.fail
import org.junit.Test

class MsProtocolClientTest {

    private suspend fun <T> withFakeEcu(
        pageSize: Int = FakeEcu.DEFAULT_PAGE_SIZE,
        block: suspend (MsProtocolClient, FakeEcu, FakeEcuTransport) -> T
    ): T {
        val ecu = FakeEcu(pageSize = pageSize)
        val transport = FakeEcuTransport(ecu)
        transport.connect()
        try {
            val client = MsProtocolClient(transport, responseTimeoutMs = 2_000)
            return block(client, ecu, transport)
        } finally {
            transport.disconnect()
        }
    }

    @Test
    fun `handshake mengembalikan signature tanpa NUL`() = runBlocking {
        withFakeEcu { client, ecu, _ ->
            assertEquals(ecu.signature, client.querySignature("Q"))
        }
    }

    @Test
    fun `handshake dengan command S juga berjalan`() = runBlocking {
        withFakeEcu { client, _, _ ->
            assertTrue(client.querySignature("S").isNotEmpty())
        }
    }

    @Test
    fun `readPage kecil sesuai isi memori fake`() = runBlocking {
        withFakeEcu { client, ecu, _ ->
            val data = client.readPage(page = 0, offset = 4, count = 16)
            assertArrayEquals(ecu.pages.getValue(0).copyOfRange(4, 20), data)
        }
    }

    @Test
    fun `readPage dipecah per blockingFactor dan di-reassembly utuh`() = runBlocking {
        withFakeEcu(pageSize = 512) { client, ecu, _ ->
            val data = client.readPage(page = 0, offset = 0, count = 300, blockingFactor = 128)
            assertArrayEquals(ecu.pages.getValue(0).copyOfRange(0, 300), data)
            val reads = ecu.receivedRequests
                .filter { it.isNotEmpty() && it[0].toInt().toChar() == 'R' }
            assertEquals(3, reads.size)
            // count per chunk: 128, 128, 44 — semua big-endian di wire
            assertEquals(listOf(128, 128, 44), reads.map { MsEnvelopeCodec.readU16BE(it, 5) })
            assertEquals(0, MsEnvelopeCodec.readU16BE(reads[1], 1))   // page
            assertEquals(128, MsEnvelopeCodec.readU16BE(reads[1], 3)) // offset chunk ke-2
        }
    }

    @Test
    fun `command tak dikenal ditolak ECU sebagai flag 0x83`() = runBlocking {
        withFakeEcu { client, _, _ ->
            try {
                client.request(byteArrayOf('Z'.code.toByte()))
                fail("Seharusnya EcuRejected")
            } catch (e: EcuCommunicationException.EcuRejected) {
                assertEquals(0x83, e.flag)
            }
        }
    }

    @Test
    fun `read melewati ujung page memicu flag 0x84`() = runBlocking {
        withFakeEcu { client, _, _ ->
            try {
                client.readPage(page = 0, offset = 250, count = 10) // 250+10 > 256
                fail("Seharusnya EcuRejected 0x84")
            } catch (e: EcuCommunicationException.EcuRejected) {
                assertEquals(0x84, e.flag)
            }
        }
    }

    @Test
    fun `transport bisu memicu timeout`() = runBlocking {
        val silent = object : EcuTransport {
            override val name = "silent"
            override val incoming: Flow<ByteArray> = flow { awaitCancellation() } // hidup tapi tak pernah bicara
            override suspend fun connect() {}
            override suspend fun disconnect() {}
            override suspend fun write(bytes: ByteArray) {}
        }
        val client = MsProtocolClient(silent, responseTimeoutMs = 150)
        try {
            client.querySignature("Q")
            fail("Seharusnya timeout")
        } catch (e: EcuCommunicationException.Timeout) {
            assertEquals(150L, e.timeoutMs)
        }
    }
}
