package com.ztune.app.presentation.connection

import android.hardware.usb.UsbDevice
import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import com.ztune.app.data.settings.AppSettings
import com.ztune.app.data.transport.EcuConnectionManager
import com.ztune.app.data.transport.EcuConnectionState
import com.ztune.app.data.transport.EcuTarget
import com.ztune.app.data.transport.UsbTransportFactory
import dagger.hilt.android.lifecycle.HiltViewModel
import java.text.SimpleDateFormat
import java.util.Date
import java.util.Locale
import javax.inject.Inject
import kotlinx.coroutines.CancellationException
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch

data class UsbDeviceInfo(val description: String, val deviceKey: String)

data class ConnectionUiState(
    val connection: EcuConnectionState = EcuConnectionState(),
    val usbDevices: List<UsbDeviceInfo> = emptyList(),
    val selectedDeviceKey: String? = null,
    val pagePreview: String? = null,
    val log: List<String> = emptyList(),
    val busy: Boolean = false
)

@HiltViewModel
class ConnectionViewModel @Inject constructor(
    private val connectionManager: EcuConnectionManager,
    private val usbFactory: UsbTransportFactory,
    private val settings: AppSettings
) : ViewModel() {

    private val _uiState = MutableStateFlow(ConnectionUiState())
    val uiState: StateFlow<ConnectionUiState> = _uiState.asStateFlow()

    private var usbDevices: List<UsbDevice> = emptyList()
    private val timeFormat = SimpleDateFormat("HH:mm:ss", Locale.US)

    init {
        log("Layar koneksi siap. Mode demo bisa dipakai tanpa hardware.")
        refreshUsbDevices()
        viewModelScope.launch {
            connectionManager.state.collect { connectionState ->
                _uiState.update { it.copy(connection = connectionState) }
            }
        }
    }

    fun refreshUsbDevices() {
        usbDevices = usbFactory.listSerialDevices()
        val infos = usbDevices.map {
            UsbDeviceInfo(description = usbFactory.describe(it), deviceKey = it.deviceId.toString())
        }
        _uiState.update { current ->
            current.copy(
                usbDevices = infos,
                selectedDeviceKey = current.selectedDeviceKey
                    ?.takeIf { key -> infos.any { d -> d.deviceKey == key } }
                    ?: infos.firstOrNull()?.deviceKey
            )
        }
        log(
            if (infos.isEmpty()) "Tidak ada device USB serial (cari CP210x/FTDI + kabel OTG)."
            else "USB serial terdeteksi: ${infos.joinToString { d -> d.description }}"
        )
    }

    fun selectDevice(key: String) = _uiState.update { it.copy(selectedDeviceKey = key) }

    fun connectUsb() {
        val key = _uiState.value.selectedDeviceKey ?: run { log("Pilih device USB dulu"); return }
        val device = usbDevices.firstOrNull { it.deviceId.toString() == key }
            ?: run { log("Device tidak ditemukan — deteksi ulang"); return }
        launchLogged("Connect USB") {
            connectionManager.connect(EcuTarget.Usb(device))
            log("Terhubung (${connectionManager.state.value.transportName})")
            val signature = connectionManager.handshake("Q")
            log("Signature ECU: \"$signature\"")
        }
    }

    fun connectDemo() {
        val scenario = settings.demoScenario.value
        launchLogged("Connect demo (${scenario.name})") {
            connectionManager.connect(EcuTarget.Demo(scenario))
            val signature = connectionManager.handshake("Q")
            log("Signature ECU (demo): \"$signature\"")
            log("Demo siap — skenario: ${scenario.name}. Buka Dashboard/Tune/Datalog.")
        }
    }

    fun checkSignature(command: String) {
        launchLogged("Handshake '$command'") {
            val signature = connectionManager.handshake(command)
            log("Signature/versi ECU: \"$signature\"")
        }
    }

    fun readPagePreview(page: Int = 0, count: Int = 256, blockingFactor: Int = 256) {
        launchLogged("Read page $page [0..$count)") {
            val data = connectionManager.withClient { client ->
                client.readPage(page = page, offset = 0, count = count, blockingFactor = blockingFactor)
            }
            _uiState.update { it.copy(pagePreview = toHexDump(data)) }
            log("OK: ${data.size} byte dibaca dari page $page")
        }
    }

    fun disconnect() {
        launchLogged("Disconnect") {
            connectionManager.disconnect()
            _uiState.update { it.copy(pagePreview = null) }
            log("Terputus dari ECU")
        }
    }

    private fun launchLogged(label: String, block: suspend () -> Unit) {
        viewModelScope.launch {
            _uiState.update { it.copy(busy = true) }
            try {
                log("$label …")
                block()
            } catch (e: CancellationException) {
                throw e
            } catch (e: Exception) {
                log("$label GAGAL: ${e.message}")
            } finally {
                _uiState.update { it.copy(busy = false) }
            }
        }
    }

    private fun log(message: String) {
        val line = "[${timeFormat.format(Date())}] $message"
        _uiState.update { it.copy(log = (it.log + line).takeLast(MAX_LOG_LINES)) }
    }

    private fun toHexDump(data: ByteArray, bytesPerLine: Int = 16): String = buildString {
        data.forEachIndexed { index, b ->
            if (index % bytesPerLine == 0) append("%04X  ".format(index))
            append("%02X ".format(b))
            if (index % bytesPerLine == bytesPerLine - 1) append('\n')
        }
    }.trimEnd()

    companion object {
        private const val MAX_LOG_LINES = 200
    }
}
