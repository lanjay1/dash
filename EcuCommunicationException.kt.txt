package com.ztune.app.domain.protocol

/** Hierarki error komunikasi ECU — UI menampilkan message, layer atas bisa bereaksi spesifik. */
sealed class EcuCommunicationException(message: String, cause: Throwable? = null) : Exception(message, cause) {

    /** Tidak ada respons dalam batas waktu. */
    class Timeout(val timeoutMs: Long) :
        EcuCommunicationException("Tidak ada respons dari ECU dalam $timeoutMs ms")

    /** Koneksi putus di tengah operasi / gagal dibuka. */
    class ConnectionClosed(cause: Throwable? = null) :
        EcuCommunicationException("Koneksi ECU bermasalah${cause?.let { ": ${it.message}" } ?: ""}", cause)

    /** Frame respons ditolak: CRC lokal tidak cocok dengan CRC di wire. */
    class CrcMismatch(val expected: Long, val actual: Long) :
        EcuCommunicationException(
            "CRC frame respons tidak cocok (hitung 0x%08X, terima 0x%08X)".format(expected, actual)
        )

    /** Struktur frame tidak valid (size di luar batas, respons pendek, dsb.). */
    class InvalidFrame(detail: String) : EcuCommunicationException("Frame tidak valid: $detail")

    /** ECU menolak command — response flag error (kompatibel MS/rusEFI). */
    class EcuRejected(val flag: Int) :
        EcuCommunicationException("ECU menolak command (flag 0x%02X: %s)".format(flag, flagName(flag))) {

        companion object {
            fun flagName(flag: Int): String = when (flag) {
                0x80 -> "underrun"
                0x82 -> "CRC fail (ECU-side)"
                0x83 -> "unrecognized command"
                0x84 -> "out of range"
                else -> "unknown"
            }
        }
    }
}
