package com.ztune.app.domain.table

/**
 * Interpolasi linear antar sel terpilih.
 * Row-wise: per baris, isi antara sel terpilih pertama & terakhir (nilai keduanya
 * jadi anchor). Column-wise: analog per kolom. Sel di luar bounding selection tak tersentuh.
 */
object TableInterpolator {

    fun rowWise(grid: TableGrid, selected: Set<CellPos>): EditBatch? {
        val edits = mutableListOf<CellEdit>()
        selected.groupBy { it.row }.forEach { (_, cells) ->
            val sorted = cells.sortedBy { it.col }
            val first = sorted.first()
            val last = sorted.last()
            if (first.col == last.col) return@forEach
            val v0 = grid.physical(first.row, first.col) ?: return@forEach
            val v1 = grid.physical(last.row, last.col) ?: return@forEach
            val span = last.col - first.col
            for (c in (first.col + 1) until last.col) {
                val t = (c - first.col).toDouble() / span
                edits += CellEdit(first.row, c, v0 + (v1 - v0) * t, v0 + (v1 - v0) * t)
            }
        }
        return if (edits.isEmpty()) null else EditBatch("Interp →", edits)
    }

    fun columnWise(grid: TableGrid, selected: Set<CellPos>): EditBatch? {
        val edits = mutableListOf<CellEdit>()
        selected.groupBy { it.col }.forEach { (_, cells) ->
            val sorted = cells.sortedBy { it.row }
            val first = sorted.first()
            val last = sorted.last()
            if (first.row == last.row) return@forEach
            val v0 = grid.physical(first.row, first.col) ?: return@forEach
            val v1 = grid.physical(last.row, last.col) ?: return@forEach
            val span = last.row - first.row
            for (r in (first.row + 1) until last.row) {
                val t = (r - first.row).toDouble() / span
                edits += CellEdit(r, first.col, v0 + (v1 - v0) * t, v0 + (v1 - v0) * t)
            }
        }
        return if (edits.isEmpty()) null else EditBatch("Interp ↓", edits)
    }
}

/** Posisi sel (baris, kolom) — dipakai selection UI & interpolasi. */
data class CellPos(val row: Int, val col: Int)
