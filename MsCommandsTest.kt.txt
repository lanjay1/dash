package com.ztune.app.domain.protocol

import org.junit.Assert.assertArrayEquals
import org.junit.Assert.assertEquals
import org.junit.Assert.assertThrows
import org.junit.Test

class MsCommandsTest {

    @Test
    fun `writeChunk bentuk payload`() {
        val p = MsCommands.writeChunk(0, 16, byteArrayOf(1, 2, 3))
        assertEquals(10, p.size)
        assertEquals('C'.code.toByte(), p[0])
        assertEquals(0, MsEnvelopeCodec.readU16BE(p, 1))
        assertEquals(16, MsEnvelopeCodec.readU16BE(p, 3))
        assertEquals(3, MsEnvelopeCodec.readU16BE(p, 5))
        assertArrayEquals(byteArrayOf(1, 2, 3), p.copyOfRange(7, 10))
    }

    @Test
    fun `burn bentuk payload`() {
        val p = MsCommands.burn(0)
        assertArrayEquals(byteArrayOf('B'.code.toByte(), 0, 0), p)
        val p1 = MsCommands.burn(1)
        assertEquals(1, MsEnvelopeCodec.readU16BE(p1, 1))
    }

    @Test
    fun `crc32Region bentuk payload`() {
        val p = MsCommands.crc32Region(0, 100, 64)
        assertEquals('k'.code.toByte(), p[0])
        assertEquals(100, MsEnvelopeCodec.readU16BE(p, 3))
        assertEquals(64, MsEnvelopeCodec.readU16BE(p, 5))
    }

    @Test
    fun `range tidak valid ditolak`() {
        assertThrows(IllegalArgumentException::class.java) { MsCommands.writeChunk(-1, 0, byteArrayOf(1)) }
        assertThrows(IllegalArgumentException::class.java) { MsCommands.burn(0x1FFFF) }
    }
}
