package com.ztune.app.domain.datalog

import java.time.Instant
import java.time.ZoneOffset
import java.time.format.DateTimeFormatter
import java.util.Locale

/**
 * Builder CSV murni (tanpa dependensi Android) — diuji unit test.
 * Format: timestamp_iso (UTC), timestamp_ms, lalu satu kolom per channel.
 * Sel kosong = nilai tidak ada/NaN pada frame tersebut.
 */
object DatalogCsv {

    private val isoFormatter: DateTimeFormatter =
        DateTimeFormatter.ofPattern("yyyy-MM-dd'T'HH:mm:ss.SSS").withZone(ZoneOffset.UTC)

    fun build(channelNames: List<String>, samples: List<DatalogSample>): String = buildString {
        append("timestamp_iso,timestamp_ms")
        channelNames.forEach { name -> append(',').append(escape(name)) }
        append('\n')
        for (sample in samples) {
            append(isoFormatter.format(Instant.ofEpochMilli(sample.timestampMs)))
            append(',')
            append(sample.timestampMs)
            for (name in channelNames) {
                append(',')
                sample.values[name]?.let { append(formatNumber(it)) }
            }
            append('\n')
        }
    }

    /** Tanpa eksponen, tanpa nol tak berarti; NaN → kosong. */
    private fun formatNumber(value: Double): String = when {
        value.isNaN() -> ""
        value == value.toLong().toDouble() -> value.toLong().toString()
        else -> String.format(Locale.US, "%.4f", value).trimEnd('0').trimEnd('.')
    }

    private fun escape(field: String): String =
        if (field.any { it == ',' || it == '"' || it == '\n' || it == '\r' }) {
            "\"" + field.replace("\"", "\"\"") + "\""
        } else {
            field
        }
}
