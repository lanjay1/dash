@OptIn(ExperimentalLayoutApi::class)
@Composable
fun DatalogScreen(
    onNavigateToConnection: () -> Unit,
    modifier: Modifier = Modifier,
    viewModel: DatalogViewModel = hiltViewModel()
) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    val exportLauncher = rememberLauncherForActivityResult(
        ActivityResultContracts.CreateDocument("text/csv")
    ) { uri -> uri?.let(viewModel::exportCsv) }

    DatalogContent(
        uiState = uiState,
        onToggleChannel = viewModel::toggleChannel,
        onStart = viewModel::startRecording,
        onStop = viewModel::stopRecording,
        onOpenSession = viewModel::openSession,
        onCloseSession = viewModel::closeSession,
        onDeleteSession = viewModel::deleteSession,
        onExport = { name -> exportLauncher.launch(name) },
        onClearMessage = viewModel::clearMessage,
        onNavigateToConnection = onNavigateToConnection,
        modifier = modifier
    )
}

@OptIn(ExperimentalLayoutApi::class)
@Composable
private fun DatalogContent(
    uiState: DatalogUiState,
    onToggleChannel: (String) -> Unit,
    onStart: () -> Unit,
    onStop: () -> Unit,
    onOpenSession: (Long) -> Unit,
    onCloseSession: () -> Unit,
    onDeleteSession: (Long) -> Unit,
    onExport: (String) -> Unit,
    onClearMessage: () -> Unit,
    onNavigateToConnection: () -> Unit,
    modifier: Modifier = Modifier
) {
    val connected = uiState.connectionStatus == EcuConnectionStatus.CONNECTED
    var confirmDelete by remember { mutableStateOf<Long?>(null) }

    Column(
        modifier = modifier
            .fillMaxSize()
            .verticalScroll(rememberScrollState())
            .padding(16.dp),
        verticalArrangement = Arrangement.spacedBy(12.dp)
    ) {
        Text(stringResource(R.string.datalog_title), style = MaterialTheme.typography.headlineMedium)
        Text(
            stringResource(R.string.datalog_subtitle),
            style = MaterialTheme.typography.bodyMedium,
            color = MaterialTheme.colorScheme.onSurfaceVariant
        )

        uiState.message?.let { message ->
            Card(modifier = Modifier.fillMaxWidth()) {
                Row(
                    modifier = Modifier.padding(horizontal = 16.dp, vertical = 8.dp),
                    verticalAlignment = Alignment.CenterVertically
                ) {
                    Text(message, modifier = Modifier.weight(1f), style = MaterialTheme.typography.bodyMedium)
                    TextButton(onClick = onClearMessage) { Text("×") }
                }
            }
        }

        if (!connected) {
            Card(modifier = Modifier.fillMaxWidth()) {
                Column(
                    modifier = Modifier.padding(16.dp),
                    verticalArrangement = Arrangement.spacedBy(8.dp)
                ) {
                    Text(stringResource(R.string.dl_not_connected), style = MaterialTheme.typography.titleMedium)
                    Button(onClick = onNavigateToConnection) {
                        Text(stringResource(R.string.dash_button_connect))
                    }
                }
            }
        } else {
            Card(modifier = Modifier.fillMaxWidth()) {
                Column(
                    modifier = Modifier.padding(16.dp),
                    verticalArrangement = Arrangement.spacedBy(8.dp)
                ) {
                    Text(stringResource(R.string.dl_record_section), style = MaterialTheme.typography.titleLarge)
                    Text(
                        stringResource(R.string.dl_channels_hint),
                        style = MaterialTheme.typography.bodyMedium,
                        color = MaterialTheme.colorScheme.onSurfaceVariant
                    )
                    FlowRow(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
                        uiState.availableChannels.forEach { option ->
                            FilterChip(
                                selected = option.name in uiState.selectedChannels,
                                enabled = !uiState.isRecording,
                                onClick = { onToggleChannel(option.name) },
                                label = {
                                    Text(
                                        if (option.units.isBlank()) option.name
                                        else "${option.name} (${option.units})"
                                    )
                                }
                            )
                        }
                    }
                    if (uiState.isRecording) {
                        Text(
                            stringResource(R.string.dl_recording_fmt, uiState.recordedCount),
                            style = MaterialTheme.typography.titleMedium.copy(fontFamily = FontFamily.Monospace),
                            color = MaterialTheme.colorScheme.primary
                        )
                    }
                    Button(
                        onClick = if (uiState.isRecording) onStop else onStart,
                        enabled = uiState.isRecording || uiState.selectedChannels.isNotEmpty(),
                        modifier = Modifier.fillMaxWidth()
                    ) {
                        Text(
                            if (uiState.isRecording) stringResource(R.string.dl_stop)
                            else stringResource(R.string.dl_start)
                        )
                    }
                }
            }
        }

        uiState.openSession?.let { session ->
            SessionDetailSection(
                session = session,
                isExporting = uiState.isExporting,
                onExport = {
                    val safeName = session.name.replace(Regex("[^A-Za-z0-9_-]+"), "_")
                    onExport("$safeName.csv")
                },
                onClose = onCloseSession,
                onAskDelete = { confirmDelete = session.id }
            )
        }

        Text(stringResource(R.string.dl_sessions_section), style = MaterialTheme.typography.titleLarge)
        if (uiState.sessions.isEmpty()) {
            Text(
                stringResource(R.string.dl_no_sessions),
                style = MaterialTheme.typography.bodyMedium,
                color = MaterialTheme.colorScheme.onSurfaceVariant
            )
        } else {
            val dateFormat = remember { SimpleDateFormat("dd MMM yyyy HH:mm", Locale.getDefault()) }
            uiState.sessions.forEach { session ->
                Card(modifier = Modifier.fillMaxWidth()) {
                    Row(
                        modifier = Modifier.padding(horizontal = 16.dp, vertical = 12.dp),
                        verticalAlignment = Alignment.CenterVertically,
                        horizontalArrangement = Arrangement.spacedBy(8.dp)
                    ) {
                        Column(modifier = Modifier.weight(1f)) {
                            Text(session.name, style = MaterialTheme.typography.titleMedium)
                            Text(
                                stringResource(
                                    R.string.dl_session_meta_fmt,
                                    dateFormat.format(Date(session.createdAtMs)), session.sampleCount
                                ),
                                style = MaterialTheme.typography.bodyMedium,
                                color = MaterialTheme.colorScheme.onSurfaceVariant
                            )
                        }
                        TextButton(onClick = { onOpenSession(session.id) }) {
                            Text(stringResource(R.string.dl_open))
                        }
                        TextButton(onClick = { confirmDelete = session.id }) {
                            Text(stringResource(R.string.dl_delete), color = MaterialTheme.colorScheme.error)
                        }
                    }
                }
            }
        }
    }

    confirmDelete?.let { target ->
        val name = uiState.sessions.firstOrNull { it.id == target }?.name
            ?: uiState.openSession?.takeIf { it.id == target }?.name ?: ""
        AlertDialog(
            onDismissRequest = { confirmDelete = null },
            title = { Text(stringResource(R.string.dl_confirm_delete_title)) },
            text = { Text(stringResource(R.string.dl_confirm_delete_body, name)) },
            confirmButton = {
                Button(onClick = { onDeleteSession(target); confirmDelete = null }) {
                    Text(stringResource(R.string.dl_confirm_ok))
                }
            },
            dismissButton = {
                TextButton(onClick = { confirmDelete = null }) { Text(stringResource(R.string.dl_cancel)) }
            }
        )
    }
}

@OptIn(ExperimentalLayoutApi::class)
@Composable
private fun SessionDetailSection(
    session: SessionDetailUi,
    isExporting: Boolean,
    onExport: () -> Unit,
    onClose: () -> Unit,
    onAskDelete: () -> Unit
) {
    var visible by remember(session.id) {
        mutableStateOf(session.channels.toSet())
    }
    val colors = remember(session.id) {
        session.channels.mapIndexed { i, n -> n to channelColor(i) }.toMap()
    }

    Card(modifier = Modifier.fillMaxWidth()) {
        Column(
            modifier = Modifier.padding(16.dp),
            verticalArrangement = Arrangement.spacedBy(8.dp)
        ) {
            Text(stringResource(R.string.dl_detail_title), style = MaterialTheme.typography.titleLarge)
            Text(session.name, style = MaterialTheme.typography.titleMedium)

            LineChart(
                samples = session.samples,
                visibleChannels = session.channels.filter { it in visible },
                colors = colors,
                modifier = Modifier
                    .fillMaxWidth()
                    .height(240.dp)
            )

            FlowRow(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
                session.channels.forEachIndexed { index, name ->
                    val color = colors[name] ?: channelColor(index)
                    FilterChip(
                        selected = name in visible,
                        onClick = {
                            visible = if (name in visible) visible - name else visible + name
                        },
                        leadingContent = {
                            Box(
                                modifier = Modifier
                                    .size(12.dp)
                                    .background(color, CircleShape)
                            )
                        },
                        label = { Text(name) }
                    )
                }
            }

            Row(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
                Button(onClick = onExport, enabled = !isExporting && session.samples.isNotEmpty()) {
                    Text(stringResource(R.string.dl_export_csv))
                }
                OutlinedButton(onClick = onClose) { Text(stringResource(R.string.dl_close)) }
                TextButton(onClick = onAskDelete) {
                    Text(stringResource(R.string.dl_delete), color = MaterialTheme.colorScheme.error)
                }
            }
        }
    }
}
