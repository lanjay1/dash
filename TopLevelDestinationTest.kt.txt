package com.ztune.app.core.navigation

import org.junit.Assert.assertEquals
import org.junit.Assert.assertTrue
import org.junit.Test

/** Smoke test navigasi: memastikan pipeline unit test berjalan sejak Fase 0. */
class TopLevelDestinationTest {

    @Test
    fun `has five top level destinations`() {
        assertEquals(5, TopLevelDestination.entries.size)
    }

    @Test
    fun `routes are unique and non-blank`() {
        val routes = TopLevelDestination.entries.map { it.route }
        assertEquals(routes.size, routes.toSet().size)
        assertTrue(routes.none { it.isBlank() })
    }

    @Test
    fun `home is the start destination route`() {
        assertEquals("home", TopLevelDestination.HOME.route)
    }
}
