package com.ztune.app.domain.protocol

/**
 * State machine reassembly frame respons dari stream byte mentah.
 * Tahan terhadap chunk terpotong di posisi mana pun (WAJIB untuk Bluetooth SPP,
 * tetap berguna untuk USB) dan melakukan resync saat menemukan Size tidak valid.
 */
class MsEnvelopeFramer(
    private val maxFrameSize: Int = DEFAULT_MAX_FRAME_SIZE
) {

    sealed interface Event {
        /** Frame lengkap & CRC valid. flag 0x00 = OK, 0x01 = realtime, >=0x80 = error ECU. */
        data class Frame(val frame: MsResponseFrame) : Event

        /** Byte dibuang (frame rusak / size tidak masuk akal); scanning tetap lanjut. */
        data class Error(val reason: String, val discardedBytes: Int) : Event
    }

    private enum class State { HEADER, BODY }

    private var state = State.HEADER
    private val header = ByteArray(2)
    private var headerPos = 0
    private var body = ByteArray(0)
    private var bodyPos = 0

    fun reset() {
        state = State.HEADER
        headerPos = 0
        bodyPos = 0
        body = ByteArray(0)
    }

    /** Masukkan chunk byte mentah; kembalikan event yang dihasilkan oleh chunk ini. */
    fun feed(chunk: ByteArray, length: Int = chunk.size): List<Event> {
        val events = mutableListOf<Event>()
        for (i in 0 until length) {
            consumeByte(chunk[i], events)
        }
        return events
    }

    private fun consumeByte(b: Byte, events: MutableList<Event>) {
        when (state) {
            State.HEADER -> {
                header[headerPos++] = b
                if (headerPos < 2) return
                val total = MsEnvelopeCodec.readU16BE(header, 0)
                val payloadLen = total - MsEnvelopeCodec.RESPONSE_HEADER_SIZE - MsEnvelopeCodec.CRC_SIZE
                if (payloadLen < 0 || payloadLen > maxFrameSize) {
                    events += Event.Error("Size frame tidak masuk akal: $total", 1)
                    // Resync: byte kedua bisa jadi byte pertama frame sebenarnya.
                    header[0] = header[1]
                    headerPos = 1
                    return
                }
                body = ByteArray(total - 2)
                bodyPos = 0
                state = State.BODY
            }

            State.BODY -> {
                body[bodyPos++] = b
                if (bodyPos < body.size) return
                emitFrame(events)
                state = State.HEADER
                headerPos = 0
            }
        }
    }

    private fun emitFrame(events: MutableList<Event>) {
        val flag = body[0].toInt() and 0xFF
        val payloadLen = body.size - 5 // flag(1) + crc(4)
        val payload = body.copyOfRange(1, 1 + payloadLen)
        val wireCrc = MsEnvelopeCodec.readCrc(body, body.size - 4)
        val calcCrc = Crc32.compute(body, 0, body.size - 4).toLong() and 0xFFFFFFFFL
        if (wireCrc != calcCrc) {
            events += Event.Error(
                "CRC frame tidak cocok (hitung 0x%08X, wire 0x%08X)".format(calcCrc, wireCrc),
                body.size + 2
            )
            return
        }
        events += Event.Frame(MsResponseFrame(flag, payload))
    }

    companion object {
        /** blockingFactor 1024 & ochBlockSize ratusan byte → 16 KB batas longgar. */
        const val DEFAULT_MAX_FRAME_SIZE = 16 * 1024
    }
}
