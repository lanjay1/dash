package com.ztune.app.domain.memory

import com.ztune.app.data.transport.FakeEcuTransport
import com.ztune.app.domain.ini.BundledEcuDefinition
import com.ztune.app.domain.ini.IniDefinitionParser
import com.ztune.app.domain.protocol.DemoChannelSimulator
import com.ztune.app.domain.protocol.EcuCommunicationException
import com.ztune.app.domain.protocol.FakeEcu
import com.ztune.app.domain.protocol.MsProtocolClient
import kotlinx.coroutines.flow.first
import kotlinx.coroutines.flow.take
import kotlinx.coroutines.flow.toList
import kotlinx.coroutines.runBlocking
import kotlinx.coroutines.withTimeout
import org.junit.Assert.assertEquals
import org.junit.Assert.assertTrue
import org.junit.Assert.fail
import org.junit.Test

class LiveDataStreamerTest {

    @Test
    fun `stream menghasilkan frame terdekode dari command O`() = runBlocking {
        val definition = IniDefinitionParser().parse(BundledEcuDefinition.INI)
        val ecu = FakeEcu(
            timeSource = { 0L },
            ochBlockSize = DemoChannelSimulator.BLOCK_SIZE,
            outputBlockProvider = { t -> DemoChannelSimulator.block(t) }
        )
        val transport = FakeEcuTransport(ecu)
        transport.connect()
        try {
            val streamer = LiveDataStreamer(MsProtocolClient(transport), definition)
            val frames = withTimeout(5_000) {
                streamer.stream(periodMs = 5).take(3).toList()
            }
            assertEquals(3, frames.size)
            frames.forEach { frame ->
                assertEquals(9, frame.values.size) // 8 channel + AFRTarget
                assertEquals(1300.0, frame.values.getValue("RPM"), 0.5)
                assertEquals(14.6, frame.values.getValue("AFR"), 0.05)
                assertEquals(14.7, frame.values.getValue("AFRTarget"), 0.05)
            }
            assertTrue(frames.zipWithNext().all { (a, b) -> b.timestampMs >= a.timestampMs })
            assertTrue(ecu.receivedRequests.any { it.isNotEmpty() && it[0].toInt().toChar() == 'O' })
        } finally {
            transport.disconnect()
        }
    }

    @Test
    fun `provider blok pendek ditolak ECU dengan flag out of range`() = runBlocking {
        val definition = IniDefinitionParser().parse(BundledEcuDefinition.INI)
        val ecu = FakeEcu(
            ochBlockSize = DemoChannelSimulator.BLOCK_SIZE,
            outputBlockProvider = { ByteArray(8) }
        )
        val transport = FakeEcuTransport(ecu)
        transport.connect()
        try {
            val streamer = LiveDataStreamer(MsProtocolClient(transport), definition)
            try {
                withTimeout(5_000) { streamer.stream(periodMs = 5).first() }
                fail("Seharusnya EcuRejected 0x84")
            } catch (e: EcuCommunicationException.EcuRejected) {
                assertEquals(0x84, e.flag)
            }
        } finally {
            transport.disconnect()
        }
    }
}
