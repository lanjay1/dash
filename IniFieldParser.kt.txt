package com.ztune.app.domain.ini

import com.ztune.app.domain.ini.model.BitRange
import com.ztune.app.domain.ini.model.ConstantDefinition
import com.ztune.app.domain.ini.model.ConstantKind
import com.ztune.app.domain.ini.model.FieldType
import com.ztune.app.domain.ini.model.IniParseIssue

/** Parser daftar field definisi konstanta: "scalar, U08, 0, \"cyl\", ..." → ConstantDefinition. */
internal object IniFieldParser {

    private val kinds = setOf("scalar", "array", "bits", "string")

    /** True jika field pertama adalah keyword kind yang dikenal. */
    fun looksLikeDefinition(firstField: String?): Boolean =
        firstField != null && firstField.trim().lowercase() in kinds

    fun parseKind(raw: String): ConstantKind = when (raw.trim().lowercase()) {
        "scalar" -> ConstantKind.SCALAR
        "array" -> ConstantKind.ARRAY
        "bits" -> ConstantKind.BITS
        else -> ConstantKind.STRING
    }

    fun parseType(raw: String): FieldType? = FieldType.fromId(raw)

    /** Parse int; dukung 0x hex. */
    fun parseInt(raw: String): Int? {
        val t = raw.trim()
        return t.toIntOrNull() ?: t.removePrefix("0x").removePrefix("0X").toIntOrNull(16)
    }

    fun parseDouble(raw: String): Double? = raw.trim().toDoubleOrNull()

    /** "[16x16]" / "[16]" / "16x16" → ArrayShape. */
    fun parseArrayShape(raw: String): ArrayShape? {
        val t = raw.trim().removePrefix("[").removeSuffix("]").trim()
        val parts = t.split("x", ignoreCase = true).map { it.trim() }
        return when (parts.size) {
            1 -> parts[0].toIntOrNull()?.let { ArrayShape(rows = 1, cols = it) }
            2 -> {
                val r = parts[0].toIntOrNull()
                val c = parts[1].toIntOrNull()
                if (r != null && c != null) ArrayShape(r, c) else null
            }
            else -> null
        }
    }

    /** "[0:2]" → BitRange(0, 2). */
    fun parseBitRange(raw: String): BitRange? {
        val t = raw.trim().removePrefix("[").removeSuffix("]").trim()
        val parts = t.split(":")
        if (parts.size != 2) return null
        val lo = parts[0].trim().toIntOrNull() ?: return null
        val hi = parts[1].trim().toIntOrNull() ?: return null
        if (lo < 0 || hi < lo) return null
        return BitRange(lo, hi)
    }

    /**
     * Parse definisi konstanta lengkap.
     * - Fail-closed (return null + issue) untuk field layout: tipe, offset, shape, panjang string.
     * - Toleran untuk field tampilan: tak bisa diparse → Double.NaN + issue, parsing lanjut.
     * - Field trailing yang hilang → default lenient (scale 1.0, translate 0.0, min/max 0.0, digits 0).
     */
    fun parseConstant(
        name: String,
        fields: List<String>,
        rawLine: String,
        page: Int,
        lineNumber: Int,
        issues: MutableList<IniParseIssue>
    ): ConstantDefinition? {
        val kind = parseKind(fields[0])
        val typeRaw = fields.getOrNull(1)?.trim().orEmpty()
        val type = parseType(typeRaw)
        if (type == null) {
            issues += IniParseIssue(lineNumber, rawLine, "Tipe field tidak dikenal: '$typeRaw'")
            return null
        }
        val offsetRaw = fields.getOrNull(2)?.trim().orEmpty()
        val offset = parseInt(offsetRaw)
        if (offset == null) {
            issues += IniParseIssue(lineNumber, rawLine, "Offset tidak valid: '$offsetRaw'")
            return null
        }

        fun issue(message: String) {
            issues += IniParseIssue(lineNumber, rawLine, message)
        }

        /** Angka opsional: default jika field tidak ada; NaN + issue jika ada tapi tak terparse. */
        fun numField(idx: Int, default: Double, label: String): Double {
            val raw = fields.getOrNull(idx) ?: return default
            return parseDouble(raw) ?: run {
                issue("Field '$label' tidak dapat diparse sebagai angka: '$raw' (ekspresi belum didukung)")
                Double.NaN
            }
        }

        return when (kind) {
            ConstantKind.SCALAR -> ConstantDefinition(
                name = name, kind = kind, type = type, page = page, offset = offset,
                units = IniTokenizer.unquote(fields.getOrNull(3) ?: ""),
                scale = numField(4, 1.0, "scale"),
                translate = numField(5, 0.0, "translate"),
                min = numField(6, 0.0, "lo"),
                max = numField(7, 0.0, "hi"),
                digits = fields.getOrNull(8)?.let { parseInt(it) } ?: 0,
                sourceLine = lineNumber
            )

            ConstantKind.ARRAY -> {
                val shapeRaw = fields.getOrNull(3)?.trim().orEmpty()
                val shape = parseArrayShape(shapeRaw)
                if (shape == null) {
                    issue("Shape array tidak valid: '$shapeRaw'")
                    return null
                }
                ConstantDefinition(
                    name = name, kind = kind, type = type, page = page, offset = offset,
                    shape = shape,
                    units = IniTokenizer.unquote(fields.getOrNull(4) ?: ""),
                    scale = numField(5, 1.0, "scale"),
                    translate = numField(6, 0.0, "translate"),
                    min = numField(7, 0.0, "lo"),
                    max = numField(8, 0.0, "hi"),
                    digits = fields.getOrNull(9)?.let { parseInt(it) } ?: 0,
                    sourceLine = lineNumber
                )
            }

            ConstantKind.BITS -> {
                val rangeRaw = fields.getOrNull(3)?.trim().orEmpty()
                val range = parseBitRange(rangeRaw)
                if (range == null) {
                    issue("Rentang bit tidak valid: '$rangeRaw'")
                    return null
                }
                ConstantDefinition(
                    name = name, kind = kind, type = type, page = page, offset = offset,
                    units = IniTokenizer.unquote(fields.getOrNull(4) ?: ""),
                    scale = numField(5, 1.0, "scale"),
                    translate = numField(6, 0.0, "translate"),
                    min = numField(7, 0.0, "lo"),
                    max = numField(8, 0.0, "hi"),
                    digits = fields.getOrNull(9)?.let { parseInt(it) } ?: 0,
                    bitRange = range,
                    labels = fields.drop(10).map { IniTokenizer.unquote(it) },
                    sourceLine = lineNumber
                )
            }

            ConstantKind.STRING -> {
                val lengthRaw = fields.getOrNull(3)?.trim().orEmpty()
                val length = parseInt(lengthRaw)
                if (length == null) {
                    issue("Panjang string tidak valid: '$lengthRaw'")
                    return null
                }
                ConstantDefinition(
                    name = name, kind = kind, type = type, page = page, offset = offset,
                    stringLength = length,
                    sourceLine = lineNumber
                )
            }
        }
    }
}
