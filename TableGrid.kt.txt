package com.ztune.app.domain.table

import com.ztune.app.domain.ini.model.ConstantDefinition
import com.ztune.app.domain.ini.model.ConstantKind
import com.ztune.app.domain.ini.model.Endianness
import com.ztune.app.domain.ini.model.FieldType
import com.ztune.app.domain.ini.model.TableDefinition
import com.ztune.app.domain.memory.ByteOrderReader
import com.ztune.app.domain.memory.ByteOrderWriter
import kotlin.math.roundToInt

/**
 * View tabel 2D atas buffer kerja page (mutable, dimiliki ViewModel).
 * Layout: ROW-MAJOR, baris mengikuti yBins, kolom mengikuti xBins
 * (grid[r][c] = data[map.offset + (r*cols + c) * elem]).
 * ⚠️ VERIFIKASI-HARDWARE: jika sumbu tertukar di rusEFI asli, ubah indexing DI SINI saja.
 */
class TableGrid(
    val table: TableDefinition,
    val map: ConstantDefinition,
    internal val data: ByteArray,
    private val byteOrder: Endianness
) {

    init {
        require(map.kind == ConstantKind.ARRAY && map.shape != null) {
            "map '${map.name}' bukan array ber-shape"
        }
        require(map.offset >= 0 && map.offset + byteSize <= data.size) {
            "Region tabel '${map.name}' (offset ${map.offset}, ${byteSize}B) di luar buffer ${data.size}B"
        }
    }

    val rows: Int = map.shape!!.rows
    val cols: Int = map.shape!!.cols
    private val elem: Int = map.type.byteSize

    val byteSize: Int get() = rows * cols * elem
    val baseOffset: Int get() = map.offset

    /** Nilai fisik sel; null jika di luar bounds atau scale/translate tak valid. */
    fun physical(row: Int, col: Int): Double? {
        if (row !in 0 until rows || col !in 0 until cols) return null
        val off = map.offset + (row * cols + col) * elem
        val raw = readRaw(off) ?: return null
        if (map.scale.isNaN() || map.translate.isNaN()) return Double.NaN
        return raw * map.scale + map.translate
    }

    /** Tulis nilai fisik ke sel (clamp ke range tipe). Return false jika di luar bounds. */
    fun setPhysical(row: Int, col: Int, value: Double): Boolean {
        if (row !in 0 until rows || col !in 0 until cols) return false
        if (map.scale.isNaN() || map.scale == 0.0) return false
        val (typeMin, typeMax) = typeRange(map.type) ?: return false
        var raw = ((value - map.translate) / map.scale).roundToInt()
        raw = raw.coerceIn(typeMin, typeMax)
        val off = map.offset + (row * cols + col) * elem
        when (map.type) {
            FieldType.U08 -> ByteOrderWriter.u8(data, off, raw)
            FieldType.S08 -> ByteOrderWriter.s8(data, off, raw)
            FieldType.U16 -> ByteOrderWriter.u16(data, off, raw, byteOrder)
            FieldType.S16 -> ByteOrderWriter.s16(data, off, raw, byteOrder)
            FieldType.U32 -> ByteOrderWriter.u32(data, off, raw.toLong(), byteOrder)
            FieldType.S32 -> ByteOrderWriter.u32(data, off, raw.toLong(), byteOrder)
            else -> return false
        }
        return true
    }

    /** Salinan byte region tabel (untuk write ke ECU). */
    fun regionBytes(): ByteArray = data.copyOfRange(map.offset, map.offset + byteSize)

    private fun readRaw(off: Int): Double? {
        val size = if (map.type == FieldType.BITS) 1 else elem
        if (off < 0 || off + size > data.size) return null
        return when (map.type) {
            FieldType.U08 -> ByteOrderReader.u8(data, off).toDouble()
            FieldType.S08 -> ByteOrderReader.s8(data, off).toDouble()
            FieldType.U16 -> ByteOrderReader.u16(data, off, byteOrder).toDouble()
            FieldType.S16 -> ByteOrderReader.s16(data, off, byteOrder).toDouble()
            FieldType.U32 -> ByteOrderReader.u32(data, off, byteOrder).toDouble()
            FieldType.S32 -> ByteOrderReader.s32(data, off, byteOrder).toDouble()
            FieldType.F32 -> ByteOrderReader.f32(data, off, byteOrder).toDouble()
            else -> null
        }
    }

    companion object {
        fun typeRange(type: FieldType): Pair<Int, Int>? = when (type) {
            FieldType.U08 -> 0 to 255
            FieldType.S08 -> -128 to 127
            FieldType.U16 -> 0 to 65535
            FieldType.S16 -> -32768 to 32767
            FieldType.U32, FieldType.S32 -> Int.MIN_VALUE to Int.MAX_VALUE
            else -> null
        }
    }
}
