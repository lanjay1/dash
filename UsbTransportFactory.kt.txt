package com.ztune.app.data.transport

import android.app.PendingIntent
import android.content.BroadcastReceiver
import android.content.Context
import android.content.Intent
import android.content.IntentFilter
import android.hardware.usb.UsbDevice
import android.hardware.usb.UsbManager
import android.os.Build
import androidx.core.content.ContextCompat
import com.hoho.android.usbserial.driver.UsbSerialProber
import com.ztune.app.domain.protocol.EcuCommunicationException
import dagger.hilt.android.qualifiers.ApplicationContext
import kotlinx.coroutines.suspendCancellableCoroutine
import kotlinx.coroutines.withTimeoutOrNull
import javax.inject.Inject
import javax.inject.Singleton
import kotlin.coroutines.resume

/**
 * Enumerasi device USB serial + alur izin USB (runtime permission khas USB Host),
 * lalu merakit [UsbSerialTransport] siap-connect().
 */
@Singleton
class UsbTransportFactory @Inject constructor(
    @ApplicationContext private val context: Context
) {

    private fun usbManager(): UsbManager =
        context.getSystemService(Context.USB_SERVICE) as UsbManager

    /** Device USB yang dikenali driver serial (CP210x/FTDI/CH340/PL2303/CDC). */
    fun listSerialDevices(): List<UsbDevice> =
        usbManager().deviceList.values
            .mapNotNull { device ->
                if (UsbSerialProber.getDefaultProber().probeDevice(device) != null) device else null
            }

    fun describe(device: UsbDevice): String {
        val driver = UsbSerialProber.getDefaultProber().probeDevice(device)
        val driverLabel = driver?.javaClass?.simpleName
            ?.removeSuffix("SerialDriver")
            ?.removeSuffix("Driver")
            ?: "unknown"
        val productName = device.productName ?: "USB Device"
        return "$productName [$driverLabel] vid=0x" + "%04X".format(device.vendorId) +
            " pid=0x" + "%04X".format(device.productId)
    }

    /**
     * Minta izin USB (jika belum), buka device, kembalikan transport siap connect().
     * Izin ditolak/timeout → EcuCommunicationException.ConnectionClosed dengan pesan jelas.
     */
    suspend fun openTransport(
        device: UsbDevice,
        baudRate: Int = UsbSerialTransport.DEFAULT_BAUD_RATE,
        permissionTimeoutMs: Long = DEFAULT_PERMISSION_TIMEOUT_MS
    ): UsbSerialTransport {
        val manager = usbManager()
        if (!manager.hasPermission(device)) {
            val granted = requestPermission(device, permissionTimeoutMs)
            if (!granted) {
                throw EcuCommunicationException.ConnectionClosed(
                    SecurityException("Izin USB untuk ${device.productName} tidak diberikan")
                )
            }
        }
        val driver = UsbSerialProber.getDefaultProber().probeDevice(device)
            ?: throw EcuCommunicationException.ConnectionClosed(
                IllegalStateException("Tidak ada driver serial untuk ${device.productName}")
            )
        val port = driver.ports.firstOrNull()
            ?: throw EcuCommunicationException.ConnectionClosed(
                IllegalStateException("Driver tidak menyediakan port: ${device.productName}")
            )
        val connection = manager.openDevice(device)
            ?: throw EcuCommunicationException.ConnectionClosed(
                IllegalStateException("Gagal membuka device USB (izin hilang / device tercabut)")
            )
        return UsbSerialTransport(
            connection = connection,
            port = port,
            baudRate = baudRate,
            name = "USB: ${describe(device)}"
        )
    }

    private suspend fun requestPermission(device: UsbDevice, timeoutMs: Long): Boolean =
        withTimeoutOrNull(timeoutMs) {
            suspendCancellableCoroutine { cont ->
                val manager = usbManager()
                if (manager.hasPermission(device)) {
                    cont.resume(true)
                    return@suspendCancellableCoroutine
                }
                val receiver = object : BroadcastReceiver() {
                    override fun onReceive(ctx: Context, intent: Intent) {
                        runCatching { context.unregisterReceiver(this) }
                        val granted = intent.getBooleanExtra(UsbManager.EXTRA_PERMISSION_GRANTED, false)
                        if (cont.isActive) cont.resume(granted)
                    }
                }
                cont.invokeOnCancellation { runCatching { context.unregisterReceiver(receiver) } }

                // FLAG_MUTABLE wajib di Android 12+ agar extras dari sistem terkirim.
                val flags = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.S) PendingIntent.FLAG_MUTABLE else 0
                val pendingIntent = PendingIntent.getBroadcast(
                    context, 0,
                    Intent(ACTION_USB_PERMISSION).setPackage(context.packageName),
                    flags
                )
                ContextCompat.registerReceiver(
                    context, receiver,
                    IntentFilter(ACTION_USB_PERMISSION),
                    ContextCompat.RECEIVER_EXPORTED
                )
                manager.requestPermission(device, pendingIntent)
            }
        } ?: false

    companion object {
        const val ACTION_USB_PERMISSION = "com.ztune.app.USB_PERMISSION"
        const val DEFAULT_PERMISSION_TIMEOUT_MS = 30_000L
    }
}
