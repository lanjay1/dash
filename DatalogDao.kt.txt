package com.ztune.app.data.db

import androidx.room.Dao
import androidx.room.Insert
import androidx.room.Query
import kotlinx.coroutines.flow.Flow

@Dao
interface DatalogDao {

    @Insert
    suspend fun insertSession(session: DatalogSessionEntity): Long

    @Insert
    suspend fun insertSamples(samples: List<DatalogSampleEntity>)

    @Query(
        """
        SELECT s.id AS id, s.name AS name, s.createdAtMs AS createdAtMs,
               (SELECT COUNT(*) FROM datalog_samples WHERE sessionId = s.id) AS sampleCount
        FROM datalog_sessions s
        ORDER BY s.createdAtMs DESC
        """
    )
    fun sessionSummaries(): Flow<List<SessionSummaryRow>>

    @Query("SELECT * FROM datalog_sessions WHERE id = :sessionId")
    suspend fun session(sessionId: Long): DatalogSessionEntity?

    @Query("SELECT * FROM datalog_samples WHERE sessionId = :sessionId ORDER BY timestampMs ASC")
    suspend fun samples(sessionId: Long): List<DatalogSampleEntity>

    @Query("DELETE FROM datalog_sessions WHERE id = :sessionId")
    suspend fun deleteSession(sessionId: Long)
}
