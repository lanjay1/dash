package com.ztune.app.domain.ini

import com.ztune.app.domain.ini.model.ConstantKind
import com.ztune.app.domain.ini.model.Endianness
import com.ztune.app.domain.ini.model.FieldType
import com.ztune.app.domain.ini.model.PAGE_NOT_MAPPED
import org.junit.Assert.assertEquals
import org.junit.Assert.assertNotNull
import org.junit.Assert.assertTrue
import org.junit.Test

class IniDefinitionParserTest {

    private fun loadFixture(): String {
        val loader = javaClass.classLoader ?: error("ClassLoader tidak tersedia")
        val stream = loader.getResourceAsStream("fixtures/minimal_ecu.ini")
        requireNotNull(stream) { "Fixture 'fixtures/minimal_ecu.ini' tidak ditemukan di test resources" }
        return stream.bufferedReader().use { it.readText() }
    }

    private fun parseFixture() = IniDefinitionParser().parse(loadFixture())

    // ---------- meta & commands ----------

    @Test
    fun `meta terbaca termasuk signature dengan titik koma di dalam kutip`() {
        val def = parseFixture()
        assertEquals("ZTUNE-TEST;001", def.meta.signature)
        assertEquals(Endianness.LITTLE, def.meta.endianness)
    }

    @Test
    fun `command serial tersimpan mentah persis seperti di ini`() {
        val def = parseFixture()
        assertEquals("S", def.commands.queryCommand)
        assertEquals("R%2i%2o%2c", def.commands.pageReadCommand)
        assertEquals("C%2i%2o%2c%v", def.commands.pageChunkWrite)
        assertEquals("B%2i", def.commands.burnCommand)
        assertEquals("k%2i%2o%2c", def.commands.crc32CheckCommand)
        assertEquals(1024, def.commands.blockingFactor)
        assertEquals("O%2o%2c", def.commands.ochGetCommand)
        assertEquals(232, def.commands.ochBlockSize)
    }

    // ---------- pages ----------

    @Test
    fun `pages terdeteksi dengan ukuran dan terurut`() {
        val def = parseFixture()
        assertEquals(listOf(1, 2), def.pages.map { it.index })
        assertEquals(256, def.pages.first { it.index == 1 }.size)
        assertEquals(64, def.pages.first { it.index == 2 }.size)
    }

    // ---------- constants ----------

    @Test
    fun `konstanta scalar array bits terparse dari fixture`() {
        val def = parseFixture()
        assertEquals(7, def.constants.size)

        val ve = def.constants["veTable1Map"]!!
        assertEquals(ConstantKind.ARRAY, ve.kind)
        assertEquals(FieldType.U08, ve.type)
        assertEquals(1, ve.page)
        assertEquals(52, ve.offset)
        assertEquals(16, ve.shape?.rows)
        assertEquals(16, ve.shape?.cols)
        assertEquals("%", ve.units)
        assertEquals(0.5, ve.scale, 1e-9)
        assertEquals(1, ve.digits)
        assertEquals(256, ve.byteSize())

        val clt = def.constants["cltBins"]!!
        assertEquals(FieldType.S08, clt.type)
        assertEquals(-40.0, clt.translate, 1e-9)

        val bits = def.constants["engineType"]!!
        assertEquals(ConstantKind.BITS, bits.kind)
        assertEquals(3, bits.bitRange?.bitCount)
        assertEquals(
            listOf("Off", "Gas", "Diesel", "Hybrid", "Electric", "Other", "Reserved", "Test"),
            bits.labels
        )

        assertEquals(32, def.constants["rpmBins"]!!.byteSize())
    }

    @Test
    fun `page switching mengarahkan konstanta ke page 2`() {
        val def = parseFixture()
        val ltft = def.constants["ltftTable"]!!
        assertEquals(2, ltft.page)
        assertEquals(0, ltft.offset)
        val warmup = def.constants["warmupClamp"]!!
        assertEquals(2, warmup.page)
        assertEquals(16, warmup.offset)
    }

    // ---------- tables ----------

    @Test
    fun `table terbaca dengan title map dan xBins yBins`() {
        val def = parseFixture()
        val t = def.tables["veTable1Tbl"]!!
        assertEquals("VE Table 1", t.title)
        assertEquals("veTable1Map", t.mapName)
        assertEquals(1, t.page)
        assertEquals("rpmBins", t.xBins?.constantName)
        assertEquals("RPMValue", t.xBins?.liveChannelName)
        assertEquals("cltBins", t.yBins?.constantName)
        assertEquals("CoolantTemp", t.yBins?.liveChannelName)
    }

    @Test
    fun `varian mapName_xBins dipasangkan ke table blok aktif`() {
        val ini = """
            [TableEditor]
               table = myTbl, myMap, "T", 1
               myMap_xBins = aBins, ChanA
               myMap_yBins = bBins, ChanB
        """.trimIndent()
        val def = IniDefinitionParser().parse(ini)
        val t = def.tables["myTbl"]!!
        assertEquals("aBins", t.xBins?.constantName)
        assertEquals("ChanA", t.xBins?.liveChannelName)
        assertEquals("bBins", t.yBins?.constantName)
    }

    @Test
    fun `varian mapName_xBins dipasangkan ke table yang sudah di-commit`() {
        val ini = """
            [TableEditor]
               table = t1, map1, "A", 1
               table = t2, map2, "B", 1
               map1_xBins = xa, ca
        """.trimIndent()
        val def = IniDefinitionParser().parse(ini)
        assertEquals("xa", def.tables["t1"]!!.xBins?.constantName)
    }

    // ---------- output channels ----------

    @Test
    fun `output channels terparse dengan page not mapped`() {
        val def = parseFixture()
        assertEquals(2, def.outputChannels.size)
        val rpm = def.outputChannels["RPM"]!!
        assertEquals(PAGE_NOT_MAPPED, rpm.page)
        assertEquals(8, rpm.offset)
        assertEquals(FieldType.U16, rpm.type)
    }

    // ---------- toleransi error ----------

    @Test
    fun `baris rusak menjadi issue dan parsing berlanjut`() {
        val def = parseFixture()
        assertEquals(1, def.issues.size)
        assertTrue(def.issues[0].message.contains("Baris tidak bisa diparse"))
        assertNotNull(def.constants["warmupClamp"]) // konstanta SETELAH baris rusak tetap terparse
    }

    @Test
    fun `assignment biasa masuk extraEntries`() {
        val def = parseFixture()
        assertEquals(1, def.extraEntries.size)
        assertEquals("chipRevision", def.extraEntries[0].key)
        assertEquals("{ 1 + 2 }", def.extraEntries[0].value)
    }

    @Test
    fun `section yang belum dimodelkan diabaikan tanpa polusi`() {
        val def = parseFixture()
        assertEquals(1, def.tables.size)
        assertTrue(def.constants.keys.none { it.startsWith("gauge") })
        assertTrue(def.outputChannels.keys.none { it.startsWith("gauge") })
    }
}
