package com.ztune.app.data.repository

import com.ztune.app.data.db.ChannelValuesCodec
import com.ztune.app.data.db.DatalogDao
import com.ztune.app.data.db.DatalogSampleEntity
import com.ztune.app.data.db.DatalogSessionEntity
import com.ztune.app.data.db.SessionSummaryRow
import com.ztune.app.domain.datalog.DatalogSample
import com.ztune.app.domain.memory.LiveDataFrame
import kotlinx.coroutines.flow.Flow
import javax.inject.Inject
import javax.inject.Singleton

data class SessionMeta(
    val id: Long,
    val name: String,
    val createdAtMs: Long,
    val channels: List<String>
)

@Singleton
class DatalogRepository @Inject constructor(
    private val dao: DatalogDao
) {

    fun sessions(): Flow<List<SessionSummaryRow>> = dao.sessionSummaries()

    suspend fun createSession(name: String, channelNames: List<String>): Long =
        dao.insertSession(
            DatalogSessionEntity(
                name = name,
                createdAtMs = System.currentTimeMillis(),
                channelNamesJson = ChannelValuesCodec.encodeNames(channelNames)
            )
        )

    /** Batch insert — dipanggil berkala selama perekaman. */
    suspend fun appendFrames(sessionId: Long, frames: List<LiveDataFrame>) {
        if (frames.isEmpty()) return
        dao.insertSamples(frames.map { frame ->
            DatalogSampleEntity(
                sessionId = sessionId,
                timestampMs = frame.timestampMs,
                valuesJson = ChannelValuesCodec.encodeValues(frame.values)
            )
        })
    }

    suspend fun sessionMeta(sessionId: Long): SessionMeta? =
        dao.session(sessionId)?.let {
            SessionMeta(it.id, it.name, it.createdAtMs, ChannelValuesCodec.decodeNames(it.channelNamesJson))
        }

    suspend fun samples(sessionId: Long): List<DatalogSample> =
        dao.samples(sessionId).map { DatalogSample(it.timestampMs, ChannelValuesCodec.decodeValues(it.valuesJson)) }

    suspend fun deleteSession(sessionId: Long) = dao.deleteSession(sessionId)
}
