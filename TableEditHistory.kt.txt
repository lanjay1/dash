package com.ztune.app.domain.table

/** Satu perubahan sel (nilai fisik). */
data class CellEdit(val row: Int, val col: Int, val oldValue: Double, val newValue: Double)

/** Unit undo/redo: edit tunggal = batch 1 sel; interpolasi = satu batch banyak sel. */
data class EditBatch(val label: String, val edits: List<CellEdit>)

/**
 * Riwayat edit atas satu [TableGrid]. Buffer kerja dimutasi langsung oleh
 * apply/undo/redo. Dirty = revision != baseRevision; baseRevision hanya
 * diperbarui setelah write ke ECU sukses.
 */
class TableEditHistory(private val maxBatches: Int = 100) {

    private val undoStack = ArrayDeque<EditBatch>()
    private val redoStack = ArrayDeque<EditBatch>()
    private var revision = 0L
    private var baseRevision = 0L

    val canUndo: Boolean get() = undoStack.isNotEmpty()
    val canRedo: Boolean get() = redoStack.isNotEmpty()
    val isDirty: Boolean get() = revision != baseRevision

    /** Terapkan batch ke grid (sel yang nilainya berubah). */
    fun apply(grid: TableGrid, batch: EditBatch) {
        val applied = mutableListOf<CellEdit>()
        for (edit in batch.edits) {
            val current = grid.physical(edit.row, edit.col) ?: continue
            if (grid.setPhysical(edit.row, edit.col, edit.newValue)) {
                applied += CellEdit(edit.row, edit.col, current, edit.newValue)
            }
        }
        if (applied.isEmpty()) return
        undoStack.addLast(batch.copy(edits = applied))
        if (undoStack.size > maxBatches) undoStack.removeFirst()
        redoStack.clear()
        revision++
    }

    fun undo(grid: TableGrid): EditBatch? {
        val batch = undoStack.removeLastOrNull() ?: return null
        for (edit in batch.edits.asReversed()) {
            grid.setPhysical(edit.row, edit.col, edit.oldValue)
        }
        redoStack.addLast(batch)
        revision--
        return batch
    }

    fun redo(grid: TableGrid): EditBatch? {
        val batch = redoStack.removeLastOrNull() ?: return null
        for (edit in batch.edits) {
            grid.setPhysical(edit.row, edit.col, edit.newValue)
        }
        undoStack.addLast(batch)
        revision++
        return batch
    }

    /** Panggil setelah write ke ECU sukses → kondisi buffer kini jadi "bersih". */
    fun markSaved() {
        baseRevision = revision
    }
}
