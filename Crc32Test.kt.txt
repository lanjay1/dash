package com.ztune.app.domain.protocol

import org.junit.Assert.assertEquals
import org.junit.Test

class Crc32Test {

    @Test
    fun `vektor standar 123456789`() {
        assertEquals(0xCBF43926u.toInt(), Crc32.compute("123456789".toByteArray()))
    }

    @Test
    fun `data kosong menghasilkan nol`() {
        assertEquals(0, Crc32.compute(ByteArray(0)))
    }

    @Test
    fun `satu byte 0x00`() {
        assertEquals(0xD202EF8Du.toInt(), Crc32.compute(byteArrayOf(0)))
    }

    @Test
    fun `konsisten dengan java util CRC32 pada data sintetis`() {
        val data = ByteArray(1024) { ((it * 31) + 7).toByte() }
        val jcrc = java.util.zip.CRC32().apply { update(data) }
        assertEquals(jcrc.value and 0xFFFFFFFFL, Crc32.compute(data).toLong() and 0xFFFFFFFFL)
    }
}
